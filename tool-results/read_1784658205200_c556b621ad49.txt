    76→/* ===================== Main view ===================== */
    77→
    78→export function LessonsView() {
    79→  const { activeSubject } = useNav()
    80→  const [topic, setTopic] = useState('')
    81→  const [subject, setSubject] = useState<string | null>(activeSubject)
    82→  const [level, setLevel] = useState<string>('Новичок')
    83→  const [lesson, setLesson] = useState<Lesson | null>(null)
    84→  const [loading, setLoading] = useState(false)
    85→  const [progress, setProgress] = useState(0)
    86→  const lessonRef = useRef<HTMLDivElement>(null)
    87→  const inputRef = useRef<HTMLTextAreaElement>(null)
    88→
    89→  // On mount: pick up handoff topic from sessionStorage (set by Paths view)
    90→  useEffect(() => {
    91→    if (typeof window === 'undefined') return
    92→    const stored = window.sessionStorage.getItem('lumina:lesson-topic')
    93→    if (stored) {
    94→      window.sessionStorage.removeItem('lumina:lesson-topic')
    95→      setTopic(stored)
    96→      // Fire generation on next tick so state settles
    97→      setTimeout(() => {
    98→        void generate(stored)
    99→      }, 60)
   100→    }
   101→    // Mount-only effect: picks up handoff topic from sessionStorage.
   102→  }, [])
   103→
   104→  // Sync subject from nav when it changes
   105→  useEffect(() => {
   106→    if (activeSubject) setSubject(activeSubject)
   107→  }, [activeSubject])
   108→
   109→  // Reading progress: track scroll position relative to lesson container
   110→  useEffect(() => {
   111→    if (!lesson) {
   112→      setProgress(0)
   113→      return
   114→    }
   115→    function onScroll() {
   116→      const el = lessonRef.current
   117→      if (!el) return
   118→      const rect = el.getBoundingClientRect()
   119→      const total = Math.max(el.offsetHeight - window.innerHeight, 1)
   120→      const scrolled = Math.max(-rect.top, 0)
   121→      const pct = Math.min(100, Math.max(0, (scrolled / total) * 100))
   122→      setProgress(pct)
   123→    }
   124→    window.addEventListener('scroll', onScroll, { passive: true })
   125→    window.addEventListener('resize', onScroll)
   126→    onScroll()
   127→    return () => {
   128→      window.removeEventListener('scroll', onScroll)
   129→      window.removeEventListener('resize', onScroll)
   130→    }
   131→  }, [lesson])
   132→
   133→  const generate = useCallback(
   134→    async (topicArg?: string) => {
   135→      const finalTopic = (topicArg ?? topic).trim()
   136→      if (!finalTopic) {
   137→        toast.error('Введи тему урока')
   138→        inputRef.current?.focus()
   139→        return
   140→      }
   141→      setTopic(finalTopic)
   142→      setLoading(true)
   143→      try {
   144→        const res = await fetch('/api/lesson', {
   145→          method: 'POST',
   146→          headers: { 'Content-Type': 'application/json' },
   147→          body: JSON.stringify({
   148→            topic: finalTopic,
   149→            subject: subject ?? 'general',
   150→            level,
   151→          }),
   152→        })
   153→        const data = await res.json()
   154→        if (!res.ok) {
   155→          throw new Error(data?.error || 'Не удалось создать урок')
   156→        }
   157→        setLesson(data.lesson as Lesson)
   158→        if (typeof data.xp === 'number') {
   159→          useUser.setState({ xp: data.xp, level: data.level })
   160→        }
   161→        toast.success('Урок готов! +25 XP')
   162→        // Scrolling is handled by the dedicated effect below (fires when lesson
   163→        // commits to the DOM, so the ref is reliably attached).
   164→      } catch (e) {
   165→        const msg = e instanceof Error ? e.message : 'Не удалось создать урок'
   166→        toast.error(msg)
   167→      } finally {
   168→        setLoading(false)
   169→      }
   170→    },
   171→    [topic, subject, level]
   172→  )
   173→
   174→  const resetToNew = useCallback(() => {
   175→    setLesson(null)
   176→    setTopic('')
   177→    setProgress(0)
   178→    setTimeout(() => inputRef.current?.focus(), 80)
   179→  }, [])
   180→
   181→  // Smooth-scroll to the lesson reader whenever a new lesson finishes loading.
   182→  // Using an effect (rather than a setTimeout inside generate) guarantees the
   183→  // ref is attached to the freshly-mounted DOM node.
   184→  useEffect(() => {
   185→    if (!lesson || loading) return
   186→    const t = window.setTimeout(() => {
   187→      lessonRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   188→    }, 80)
   189→    return () => window.clearTimeout(t)
   190→  }, [lesson, loading])
   191→
   192→  const subjectObj = subject ? getSubject(subject) : undefined
   193→
   194→  return (
   195→    <PageSection className="py-8">
   196→      {/* Reading progress bar — fixed below the topbar */}
   197→      <AnimatePresence>
   198→        {lesson && (
   199→          <motion.div
   200→            key="progress-bar"
   201→            initial={{ opacity: 0 }}
   202→            animate={{ opacity: 1 }}
   203→            exit={{ opacity: 0 }}
   204→            className="fixed left-0 right-0 top-16 z-30 h-1"
   205→          >
   206→            <div
   207→              className="h-full bg-gradient-to-r from-violet-500 via-fuchsia-500 to-pink-500 transition-[width] duration-150 ease-out"
   208→              style={{ width: `${progress}%` }}
   209→            />
   210→          </motion.div>
   211→        )}
   212→      </AnimatePresence>
   213→
   214→      <SectionHeader
   215→        title="Интерактивные уроки"
   216→        subtitle="Введи тему — AI создаст красивый урок за секунды"
   217→        icon={BookOpen}
   218→      />
   219→
   220→      {/* Input panel: full or compact */}
   221→      {/* Saved lessons */}
   222→      <SavedLessonsBar onOpen={(t, s) => { setTopic(t); setSubject(s); generate(t) }} />
   223→
   224→      <div className="mb-6">
   225→        <AnimatePresence mode="wait">
   226→          {lesson ? (
   227→            <motion.div
   228→              key="compact-bar"
   229→              initial={{ opacity: 0, y: -8 }}
   230→              animate={{ opacity: 1, y: 0 }}
   231→              exit={{ opacity: 0, y: -8 }}
   232→              transition={{ duration: 0.2 }}
   233→            >
   234→              <CompactBar
   235→                lesson={lesson}
   236→                onNew={resetToNew}
   237→                subjectLabel={subjectObj?.ru}
   238→                onScrollToLesson={() =>
   239→                  lessonRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   240→                }
   241→              />
   242→            </motion.div>
   243→          ) : (
   244→            <motion.div
   245→              key="full-input"
   246→              initial={{ opacity: 0, y: -8 }}
   247→              animate={{ opacity: 1, y: 0 }}
   248→              exit={{ opacity: 0, y: -8 }}
   249→              transition={{ duration: 0.2 }}
   250→            >
   251→              <FullInputPanel
   252→                topic={topic}
   253→                setTopic={setTopic}
   254→                subject={subject}
   255→                setSubject={setSubject}
   256→                level={level}
   257→                setLevel={setLevel}
   258→                loading={loading}
   259→                onGenerate={generate}
   260→                inputRef={inputRef}
   261→              />
   262→            </motion.div>
   263→          )}
   264→        </AnimatePresence>
   265→      </div>
   266→
   267→      {/* Skeleton while generating */}
   268→      <AnimatePresence>
   269→        {loading && (
   270→          <motion.div
   271→            key="skeleton"
   272→            initial={{ opacity: 0, y: 8 }}
   273→            animate={{ opacity: 1, y: 0 }}
   274→            exit={{ opacity: 0 }}
   275→            transition={{ duration: 0.25 }}
   276→          >
   277→            <LessonSkeleton />
   278→          </motion.div>
   279→        )}
   280→      </AnimatePresence>
   281→
   282→      {/* Lesson reader — keyed by title so a new lesson re-mounts & replays
   283→          the reveal animation. No AnimatePresence here so the new node mounts
   284→          immediately (keeping the scroll-to-lesson ref reliable). */}
   285→      {lesson && !loading && (
   286→        <motion.div
   287→          key={`reader-${lesson.title}`}
   288→          ref={lessonRef}
   289→          initial={{ opacity: 0, y: 12 }}
   290→          animate={{ opacity: 1, y: 0 }}
   291→          transition={{ duration: 0.45, ease: 'easeOut' }}
   292→        >
   293→          <LessonReader
   294→            lesson={lesson}
   295→            subjectLabel={subjectObj?.ru}
   296→            onNext={(t) => generate(t)}
   297→          />
   298→        </motion.div>
   299→      )}
   300→
   301→      {/* Empty state with suggested topics */}
   302→      <AnimatePresence>
   303→        {!lesson && !loading && (
   304→          <motion.div
   305→            key="empty"
   306→            initial={{ opacity: 0 }}
   307→            animate={{ opacity: 1 }}
   308→            exit={{ opacity: 0 }}
   309→          >
   310→            <EmptyState
   311→              icon={Sparkles}
   312→              title="Идеи для изучения"
   313→              description="Введи свою тему выше или начни с одной из этих"
   314→              action={
   315→                <div className="mt-2 flex flex-wrap justify-center gap-2">
   316→                  {SUGGESTED_TOPICS.map((t) => (
   317→                    <button
   318→                      key={t}
   319→                      onClick={() => generate(t)}
   320→                      className="group inline-flex items-center gap-1.5 rounded-full border border-border/60 bg-background/60 px-3.5 py-1.5 text-xs font-medium text-muted-foreground transition-all hover:border-primary/40 hover:text-foreground"
   321→                    >
   322→                      {t}
   323→                      <ChevronRight className="h-3 w-3 -translate-x-1 opacity-0 transition-all group-hover:translate-x-0 group-hover:opacity-100" />
   324→                    </button>
   325→                  ))}
   326→                </div>
   327→              }
   328→            />
   329→          </motion.div>
   330→        )}
   331→      </AnimatePresence>
   332→    </PageSection>
   333→  )
   334→}

   336→/* ===================== Full input panel ===================== */
   337→
   338→function FullInputPanel({
   339→  topic,
   340→  setTopic,
   341→  subject,
   342→  setSubject,
   343→  level,
   344→  setLevel,
   345→  loading,
   346→  onGenerate,
   347→  inputRef,
   348→}: {
   349→  topic: string
   350→  setTopic: (v: string) => void
   351→  subject: string | null
   352→  setSubject: (v: string | null) => void
   353→  level: string
   354→  setLevel: (v: string) => void
   355→  loading: boolean
   356→  onGenerate: (topic?: string) => void
   357→  inputRef: React.RefObject<HTMLTextAreaElement | null>
   358→}) {
   359→  return (
   360→    <GlassCard className="overflow-hidden" hover={false} gradient>
   361→      <div className="p-5 sm:p-6">
   362→        <label className="mb-2 block text-sm font-medium">Что хочешь изучить?</label>
   363→        <textarea
   364→          ref={inputRef}
   365→          value={topic}
   366→          onChange={(e) => setTopic(e.target.value)}
   367→          placeholder="Например: как работают нейронные сети"
   368→          rows={2}
   369→          onKeyDown={(e) => {
   370→            if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   371→              e.preventDefault()
   372→              onGenerate()
   373→            }
   374→          }}
   375→          className="w-full resize-none rounded-xl border border-border bg-background/60 px-4 py-3 text-base outline-none ring-primary/20 transition placeholder:text-muted-foreground/60 focus:ring-2"
   376→        />
   377→
   378→        {/* Subject chips */}
   379→        <div className="mt-4">
   380→          <p className="mb-2 text-xs text-muted-foreground">Предмет:</p>
   381→          <div className="flex flex-wrap gap-2">
   382→            <SubjectChip
   383→              active={!subject}
   384→              emoji="✨"
   385→              label="Авто"
   386→              onClick={() => setSubject(null)}
   387→            />
   388→            {SUBJECTS.map((s) => (
   389→              <SubjectChip
   390→                key={s.id}
   391→                active={subject === s.id}
   392→                emoji={s.emoji}
   393→                label={s.ru}
   394→                onClick={() => setSubject(s.id)}
   395→              />
   396→            ))}
   397→          </div>
   398→        </div>
   399→
   400→        {/* Level + Generate */}
   401→        <div className="mt-4 flex flex-wrap items-center justify-between gap-3">
   402→          <div>
   403→            <p className="mb-1.5 text-xs text-muted-foreground">Уровень:</p>
   404→            <div className="inline-flex items-center gap-1 rounded-lg border border-border/60 bg-muted/30 p-1">
   405→              {LEVELS.map((l) => (
   406→                <button
   407→                  key={l}
   408→                  onClick={() => setLevel(l)}
   409→                  className={`rounded-md px-3 py-1 text-xs font-medium transition-colors ${
   410→                    level === l
   411→                      ? 'bg-primary text-primary-foreground shadow-sm'
   412→                      : 'text-muted-foreground hover:text-foreground'
   413→                  }`}
   414→                >
   415→                  {l}
   416→                </button>
   417→              ))}
   418→            </div>
   419→          </div>
   420→          <GradientButton onClick={() => onGenerate()} disabled={loading} className="mt-5 sm:mt-0">
   421→            {loading ? (
   422→              <>
   423→                <Loader2 className="h-4 w-4 animate-spin" /> Создаю урок...
   424→              </>
   425→            ) : (
   426→              <>
   427→                <Sparkles className="h-4 w-4" /> Создать урок
   428→              </>
   429→            )}
   430→          </GradientButton>
   431→        </div>
   432→
   433→        <p className="mt-3 text-[11px] text-muted-foreground/70">
   434→          Подсказка: нажми <kbd className="rounded border border-border/60 bg-muted/40 px-1 py-0.5 font-mono text-[10px]">⌘/Ctrl + Enter</kbd> для быстрого запуска
   435→        </p>
   436→      </div>
   437→    </GlassCard>
   438→  )
   439→}

   559→/* ===================== Lesson reader ===================== */
   560→
   561→function LessonReader({
   562→  lesson,
   563→  subjectLabel,
   564→  onNext,
   565→}: {
   566→  lesson: Lesson
   567→  subjectLabel?: string
   568→  onNext: (topic: string) => void
   569→}) {
   570→  const blocks = lesson.blocks ?? []
   571→  const takeaways = lesson.keyTakeaways ?? []
   572→
   573→  return (
   574→    <GlassCard className="overflow-hidden" hover={false}>
   575→      {/* Hero header */}
   576→      <div className="relative overflow-hidden border-b border-border/60">
   577→        <div className="absolute inset-0 bg-gradient-to-br from-violet-500/15 via-fuchsia-500/10 to-pink-500/15" />
   578→        <div className="absolute inset-0 bg-grid opacity-15" />
   579→        <div className="relative p-6 sm:p-8">
   580→          <div className="flex flex-wrap items-start gap-4 sm:gap-5">
   581→            <motion.div
   582→              initial={{ scale: 0.7, rotate: -10, opacity: 0 }}
   583→              animate={{ scale: 1, rotate: 0, opacity: 1 }}
   584→              transition={{ duration: 0.6, ease: [0.34, 1.56, 0.64, 1] }}
   585→              className="grid h-20 w-20 shrink-0 place-items-center rounded-3xl bg-gradient-to-br from-violet-500 via-fuchsia-500 to-pink-500 text-5xl shadow-lg shadow-fuchsia-500/30"
   586→            >
   587→              {lesson.emoji || '📖'}
   588→            </motion.div>
   589→            <div className="min-w-0 flex-1">
   590→              <h1 className="text-balance text-2xl font-extrabold tracking-tight sm:text-3xl lg:text-4xl">
   591→                {lesson.title}
   592→              </h1>
   593→              <p className="mt-2 text-pretty text-sm text-muted-foreground sm:text-base">
   594→                {lesson.summary}
   595→              </p>
   596→              <div className="mt-3 flex flex-wrap gap-2">
   597→                <Pill>
   598→                  <Clock className="h-3 w-3" /> {lesson.durationMin} мин
   599→                </Pill>
   600→                <Pill>
   601→                  <Gauge className="h-3 w-3" /> {lesson.difficulty}
   602→                </Pill>
   603→                {subjectLabel && (
   604→                  <Pill>
   605→                    <BookOpen className="h-3 w-3" /> {subjectLabel}
   606→                  </Pill>
   607→                )}
   608→              </div>
   609→
   610→              {/* Media actions: audio narration + AI illustration + bookmark */}
   611→              <div className="mt-4 flex flex-wrap items-center gap-2">
   612→                <AudioNarration
   613→                  text={`${lesson.title}. ${lesson.summary}. ${takeaways.join(' ')}`}
   614→                  label="Слушать урок"
   615→                />
   616→                <LessonIllustration topic={lesson.title} summary={lesson.summary} />
   617→                <BookmarkButton topic={lesson.title} subject={subjectLabel ?? 'general'} lessonJson={JSON.stringify(lesson)} />
   618→              </div>
   619→            </div>
   620→          </div>
   621→        </div>
   622→      </div>
   623→
   624→      {/* Blocks */}
   625→      <div className="p-5 sm:p-8">
   626→        <StaggerGroup className="space-y-5">
   627→          {blocks.map((block, i) => (
   628→            <StaggerItem key={i}>
   629→              <BlockRenderer block={block} />
   630→            </StaggerItem>
   631→          ))}
   632→        </StaggerGroup>
   633→
   634→        {/* Key takeaways */}
   635→        {takeaways.length > 0 && (
   636→          <div className="mt-10">
   637→            <div className="mb-3 flex items-center gap-2.5">
   638→              <div className="grid h-8 w-8 place-items-center rounded-lg bg-gradient-to-br from-violet-500/20 to-fuchsia-500/20 text-primary ring-1 ring-primary/20">
   639→                <CheckCircle2 className="h-4 w-4" />
   640→              </div>
   641→              <h3 className="text-lg font-bold tracking-tight">Ключевые выводы</h3>
   642→            </div>
   643→            <StaggerGroup className="grid gap-3 sm:grid-cols-2">
   644→              {takeaways.map((kt, i) => (
   645→                <StaggerItem key={i}>
   646→                  <div className="flex items-start gap-2.5 rounded-xl border border-border/60 bg-background/40 p-3 transition-colors hover:border-primary/30">
   647→                    <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0 text-emerald-400" />
   648→                    <span className="text-sm leading-relaxed text-muted-foreground">{kt}</span>
   649→                  </div>
   650→                </StaggerItem>
   651→              ))}
   652→            </StaggerGroup>
   653→          </div>
   654→        )}
   655→
   656→        {/* What's next */}
   657→        {lesson.nextTopic && (
   658→          <motion.div
   659→            initial={{ opacity: 0, y: 12 }}
   660→            whileInView={{ opacity: 1, y: 0 }}
   661→            viewport={{ once: true, margin: '-80px' }}
   662→            transition={{ duration: 0.4 }}
   663→            className="mt-10 overflow-hidden rounded-2xl border border-primary/20 bg-gradient-to-br from-violet-500/10 via-fuchsia-500/10 to-pink-500/10 p-5"
   664→          >
   665→            <p className="mb-1 text-xs font-semibold uppercase tracking-wider text-primary/80">
   666→              Что дальше
   667→            </p>
   668→            <div className="flex flex-wrap items-center justify-between gap-3">
   669→              <p className="text-base font-semibold sm:text-lg">{lesson.nextTopic}</p>
   670→              <GradientButton onClick={() => onNext(lesson.nextTopic)}>
   671→                Создать урок <ArrowRight className="h-4 w-4" />
   672→              </GradientButton>
   673→            </div>
   674→          </motion.div>
   675→        )}
   676→      </div>
   677→    </GlassCard>
   678→  )
   679→}

   681→/* ===================== Block renderer ===================== */
   682→
   683→function BlockRenderer({ block }: { block: LessonBlock }) {
   684→  switch (block.type) {
   685→    case 'heading':
   686→      return (
   687→        <div className="pt-2">
   688→          <h2 className="text-xl font-bold tracking-tight sm:text-2xl">{block.text}</h2>
   689→          <div className="mt-2 h-0.5 w-16 rounded-full bg-gradient-to-r from-violet-500 to-fuchsia-500" />
   690→        </div>
   691→      )
   692→
   693→    case 'paragraph':
   694→      return (
   695→        <MarkdownText className="text-[15px] text-foreground/90">{block.text}</MarkdownText>
   696→      )
   697→
   698→    case 'analogy':
   699→      return (
   700→        <AccentCard
   701→          icon={Lightbulb}
   702→          label="Аналогия"
   703→          accent="amber"
   704→        >
   705→          <MarkdownText className="text-sm text-foreground/90">{block.text}</MarkdownText>
   706→        </AccentCard>
   707→      )
   708→
   709→    case 'example':
   710→      return (
   711→        <AccentCard
   712→          icon={FlaskConical}
   713→          label="Пример"
   714→          accent="emerald"
   715→        >
   716→          <MarkdownText className="text-sm text-foreground/90">{block.text}</MarkdownText>
   717→        </AccentCard>
   718→      )
   719→
   720→    case 'callout':
   721→      return (
   722→        <AccentCard
   723→          icon={AlertCircle}
   724→          label="Важно"
   725→          accent="rose"
   726→        >
   727→          <MarkdownText className="text-sm text-foreground/90">{block.text}</MarkdownText>
   728→        </AccentCard>
   729→      )
   730→
   731→    case 'code':
   732→      return (
   733→        <div className="overflow-hidden rounded-2xl border border-border/60 bg-zinc-950/90">
   734→          {block.lang && (
   735→            <div className="flex items-center gap-2 border-b border-white/5 bg-white/[0.03] px-4 py-2">
   736→              <Code2 className="h-3.5 w-3.5 text-muted-foreground" />
   737→              <span className="font-mono text-xs text-muted-foreground">{block.lang}</span>
   738→            </div>
   739→          )}
   740→          <pre className="overflow-x-auto p-4 text-sm leading-relaxed">
   741→            <code className="font-mono text-foreground/90">{block.code ?? ''}</code>
   742→          </pre>
   743→        </div>
   744→      )
   745→
   746→    case 'steps':
   747→      return (
   748→        <div className="rounded-2xl border border-border/60 bg-background/40 p-4 sm:p-5">
   749→          <div className="mb-3 flex items-center gap-2">
   750→            <ListOrdered className="h-4 w-4 text-primary" />
   751→            <p className="text-sm font-semibold">Шаги</p>
   752→          </div>
   753→          <ol className="relative space-y-3 before:absolute before:left-[15px] before:top-2 before:h-[calc(100%-1rem)] before:w-px before:bg-border">
   754→            {(block.items ?? []).map((item, i) => (
   755→              <li key={i} className="relative flex gap-3">
   756→                <div className="z-10 grid h-8 w-8 shrink-0 place-items-center rounded-full border-2 border-primary/40 bg-card text-xs font-bold text-primary">
   757→                  {i + 1}
   758→                </div>
   759→                <div className="flex-1 pt-1">
   760→                  <MarkdownText className="text-sm text-foreground/90">{item}</MarkdownText>
   761→                </div>
   762→              </li>
   763→            ))}
   764→          </ol>
   765→        </div>
   766→      )
   767→
   768→    case 'quote':
   769→      return (
   770→        <div className="relative overflow-hidden rounded-2xl border border-border/60 bg-background/40 p-5 sm:p-6">
   771→          <Quote className="absolute right-4 top-3 h-10 w-10 text-primary/15" />
   772→          <blockquote className="relative text-base italic leading-relaxed text-foreground/90 sm:text-lg">
   773→            {block.text}
   774→          </blockquote>
   775→        </div>
   776→      )
   777→
   778→    default:
   779→      return null
   780→  }
   781→}

   850→/* ===================== Saved Lessons Bar ===================== */
   851→
   852→function SavedLessonsBar({ onOpen }: { onOpen: (topic: string, subject: string) => void }) {
   853→  const [bookmarks, setBookmarks] = useState<Array<{ id: string; topic: string; subject: string; createdAt: string }>>([])
   854→  const [open, setOpen] = useState(false)
   855→
   856→  useEffect(() => {
   857→    fetch('/api/bookmarks', { cache: 'no-store' })
   858→      .then((r) => r.json())
   859→      .then((data) => setBookmarks(data.bookmarks ?? []))
   860→      .catch(() => {})
   861→  }, [])
   862→
   863→  if (bookmarks.length === 0) return null
   864→
   865→  return (
   866→    <div className="mb-4">
   867→      <button
   868→        onClick={() => setOpen(!open)}
   869→        className="flex items-center gap-2 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   870→      >
   871→        <Bookmark className="h-4 w-4 text-primary" />
   872→        Избранные уроки ({bookmarks.length})
   873→        {open ? <ChevronRight className="h-3.5 w-3.5 rotate-90" /> : <ChevronRight className="h-3.5 w-3.5" />}
   874→      </button>
   875→      <AnimatePresence>
   876→        {open && (
   877→          <motion.div
   878→            initial={{ height: 0, opacity: 0 }}
   879→            animate={{ height: 'auto', opacity: 1 }}
   880→            exit={{ height: 0, opacity: 0 }}
   881→            transition={{ duration: 0.25 }}
   882→            className="overflow-hidden"
   883→          >
   884→            <div className="mt-2 flex flex-wrap gap-2">
   885→              {bookmarks.map((b) => (
   886→                <button
   887→                  key={b.id}
   888→                  onClick={() => onOpen(b.topic, b.subject)}
   889→                  className="inline-flex items-center gap-1.5 rounded-lg border border-border/60 bg-card/60 px-3 py-1.5 text-xs font-medium text-muted-foreground transition-colors hover:border-primary/40 hover:text-foreground"
   890→                >
   891→                  <Bookmark className="h-3 w-3 fill-primary text-primary" />
   892→                  {b.topic}
   893→                </button>
   894→              ))}
   895→            </div>
   896→          </motion.div>
   897→        )}
   898→      </AnimatePresence>
   899→    </div>
   900→  )
   901→}
