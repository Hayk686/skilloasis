   195→   195→function build7DaySeries(byDay: Record<string, number>): DaySeriesItem[] {
   196→   196→  const days: DaySeriesItem[] = []
   197→   197→  const today = new Date()
   198→   198→  for (let i = 6; i >= 0; i--) {
   199→   199→    const d = new Date(today)
   200→   200→    d.setDate(today.getDate() - i)
   201→   201→    const key = d.toISOString().slice(0, 10)
   202→   202→    const weekday = d.toLocaleDateString('ru-RU', { weekday: 'short' })
   203→   203→    const isToday = i === 0
   204→   204→    days.push({
   205→   205→      key,
   206→   206→      label: weekday.charAt(0).toUpperCase() + weekday.slice(1),
   207→   207→      count: byDay[key] || 0,
   208→   208→      isToday,
   209→   209→    })
   210→   210→  }
   211→   211→  return days
   212→   212→}

   315→   315→
   316→   316→function ConfettiBurst({ burst }: { burst: number }) {
   317→   317→  const particles = useMemo(() => {
   318→   318→    if (burst === 0) return []
   319→   319→    return Array.from({ length: 28 }, (_, i) => {
   320→   320→      const angle = (i / 28) * Math.PI * 2 + Math.random() * 0.4
   321→   321→      const distance = 90 + Math.random() * 110
   322→   322→      return {
   323→   323→        id: `${burst}-${i}`,
   324→   324→        x: Math.cos(angle) * distance,
   325→   325→        y: Math.sin(angle) * distance,
   326→   326→        delay: Math.random() * 0.12,
   327→   327→        emoji: ['🎉', '✨', '⭐', '🔥', '💫', '🌟'][i % 6],
   328→   328→        rotate: (Math.random() - 0.5) * 540,
   329→   329→      }
   330→   330→    })
   331→   331→  }, [burst])
   332→   332→
   333→   333→  return (
   334→   334→    <div className="pointer-events-none absolute inset-0 flex items-center justify-center overflow-visible">
   335→   335→      <AnimatePresence>
   336→   336→        {particles.map((p) => (
   337→   337→          <motion.div
   338→   338→            key={p.id}
   339→   339→            className="absolute text-lg"
   340→   340→            initial={{ x: 0, y: 0, opacity: 1, scale: 0, rotate: 0 }}
   341→   341→            animate={{
   342→   342→              x: p.x,
   343→   343→              y: p.y,
   344→   344→              opacity: 0,
   345→   345→              scale: 1.1,
   346→   346→              rotate: p.rotate,
   347→   347→            }}
   348→   348→            transition={{ duration: 1.5, delay: p.delay, ease: 'easeOut' }}
   349→   349→          >
   350→   350→            {p.emoji}
   351→   351→          </motion.div>
   352→   352→        ))}
   353→   353→      </AnimatePresence>
   354→   354→    </div>
   355→   355→  )
   356→   356→}

   447→   447→function ChartTooltip({ active, payload, label }: TooltipProps<number, string>) {
   448→   448→  if (!active || !payload || payload.length === 0) return null
   449→   449→  const v = Number(payload[0]?.value ?? 0)
   450→   450→  return (
   451→   451→    <div className="rounded-xl border border-border/70 bg-card/95 px-3 py-2 text-xs shadow-xl backdrop-blur-md">
   452→   452→      <div className="font-semibold text-foreground">{label}</div>
   453→   453→      <div className="mt-0.5 text-muted-foreground">
   454→   454→        {v === 0 ? 'нет активности' : `${v} ${pluralActivity(v)}`}
   455→   455→      </div>
   456→   456→    </div>
   457→   457→  )
   458→   458→}
   459→   459→
   460→   460→function pluralActivity(n: number) {
   461→   461→  const mod10 = n % 10
   462→   462→  const mod100 = n % 100
   463→   463→  if (mod10 === 1 && mod100 !== 11) return 'действие'
   464→   464→  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 10 || mod100 >= 20))
   465→   465→    return 'действия'
   466→   466→  return 'действий'
   467→   467→}
