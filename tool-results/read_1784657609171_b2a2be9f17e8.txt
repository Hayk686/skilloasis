   120→   120→  // Generate form state

   124→   124→      ? activeSubject

   127→   127→  const [count, setCount] = useState<number>(8)

   129→   129→

   138→   138→  const [flipped, setFlipped] = useState(false)

   140→   140→  const [reviewedCount, setReviewedCount] = useState(0)

   166→   166→      setLoadingDue(false)

   176→   176→  async function handleGenerate(givenTopic?: string) {
   177→   177→    const t = (givenTopic ?? topic).trim()
   178→   178→    if (!t) {
   179→   179→      toast.error('Введи тему для карточек')
   180→   180→      return
   181→   181→    }
   182→   182→    setTopic(t)
   183→   183→    setGenerating(true)
   184→   184→    try {
   185→   185→      const res = await fetch('/api/flashcards', {
   186→   186→        method: 'POST',
   187→   187→        headers: { 'Content-Type': 'application/json' },
   188→   188→        body: JSON.stringify({ topic: t, subject, count }),
   189→   189→      })
   190→   190→      const data = await res.json()
   191→   191→      if (!res.ok) throw new Error(data.error || 'Ошибка генерации')
   192→   192→      if (!data.cards?.length) throw new Error('Карты не сгенерированы')
   193→   193→
   194→   194→      useUser.setState({
   195→   195→        xp: typeof data.xp === 'number' ? data.xp : user.xp,
   196→   196→        level: typeof data.level === 'number' ? data.level : user.level,
   197→   197→      })
   198→   198→
   199→   199→      toast.success(`Создано ${data.cards.length} карточек!`, {
   200→   200→        description: 'Начинаем сессию повторения',
   201→   201→      })
   202→   202→
   203→   203→      // The POST returns cards without ids. Refetch due list to get the
   204→   204→      // persisted versions (newly created cards default dueAt = now).
   205→   205→      const dueRes = await fetch('/api/flashcards?limit=50', { cache: 'no-store' })
   206→   206→      const dueData = await dueRes.json()
   207→   207→      const all: FlashCard[] = Array.isArray(dueData.cards) ? dueData.cards : []
   208→   208→      setDueCards(all)
   209→   209→
   210→   210→      // Match persisted cards by front text to ensure ids are present.
   211→   211→      const generatedFronts = new Set(
   212→   212→        (data.cards as FlashCard[]).map((c) => c.front.trim())
   213→   213→      )
   214→   214→      let session: FlashCard[] = all.filter((c) =>
   215→   215→        generatedFronts.has((c.front || '').trim())
   216→   216→      )
   217→   217→      // Fallback: if matching failed for some reason, use the generated
   218→   218→      // payloads directly (PATCH will be skipped for cards without ids).
   219→   219→      if (session.length !== data.cards.length) {
   220→   220→        session = data.cards as FlashCard[]
   221→   221→      }
   222→   222→
   223→   223→      startSession(session, 'generated')
   224→   224→    } catch (e) {
   225→   225→      console.error('[flashcards] generate', e)
   226→   226→      toast.error('Не удалось создать карты. Попробуй ещё раз.')
   227→   227→    } finally {
   228→   228→      setGenerating(false)
   229→   229→    }
   230→   230→  }
   231→   231→
   232→   232→  /* ---------------------------- session ---------------------------- */
   233→   233→
   234→   234→  function startSession(cards: FlashCard[], source: SessionSource) {
   235→   235→    if (!cards.length) {
   236→   236→      toast.error('Нет карточек для повторения')
   237→   237→      return
   238→   238→    }
   239→   239→    setSessionCards(cards)
   240→   240→    setSessionSource(source)
   241→   241→    setCurrentIndex(0)
   242→   242→    setFlipped(false)
   243→   243→    setDirection(1)
   244→   244→    setReviewedCount(0)
   245→   245→    setXpGained(0)
   246→   246→    setSubmitting(false)
   247→   247→    setDeleting(false)
   248→   248→    setMode('review')
   249→   249→  }
   250→   250→
   251→   251→  function startDueReview() {
   252→   252→    if (!dueCards.length) return
   253→   253→    startSession(dueCards, 'due')
   254→   254→  }
   255→   255→
   256→   256→  function flip() {
   257→   257→    setFlipped((f) => !f)
   258→   258→  }
   259→   259→  flipRef.current = flip
   260→   260→
   261→   261→  async function submitQuality(quality: Quality) {
   262→   262→    if (submitting || deleting) return
   263→   263→    const card = sessionCards[currentIndex]
   264→   264→    if (!card) return
   265→   265→
   266→   266→    // No id (e.g. generated fallback) — just advance.
   267→   267→    if (!card.id) {
   268→   268→      setReviewedCount((v) => v + 1)
   269→   269→      advance()
   270→   270→      return
   271→   271→    }
   272→   272→
   273→   273→    setSubmitting(true)
   274→   274→    try {
   275→   275→      const res = await fetch(`/api/flashcards/${card.id}`, {
   276→   276→        method: 'PATCH',
   277→   277→        headers: { 'Content-Type': 'application/json' },
   278→   278→        body: JSON.stringify({ quality }),
   279→   279→      })
   280→   280→      const data = await res.json()
   281→   281→      if (!res.ok) throw new Error(data.error || 'Ошибка')
   282→   282→
   283→   283→      useUser.setState({
   284→   284→        xp: typeof data.xp === 'number' ? data.xp : user.xp,
   285→   285→        level: typeof data.level === 'number' ? data.level : user.level,
   286→   286→      })
   287→   287→
   288→   288→      let gain = 0
   289→   289→      if (quality === 'easy') gain = 5
   290→   290→      else if (quality === 'good') gain = 3
   291→   291→      setXpGained((v) => v + gain)
   292→   292→      setReviewedCount((v) => v + 1)
   293→   293→
   294→   294→      advance()
   295→   295→    } catch (e) {
   296→   296→      console.error('[flashcards] submit', e)
   297→   297→      toast.error('Не удалось сохранить ответ')
   298→   298→    } finally {
   299→   299→      setSubmitting(false)
   300→   300→    }
   301→   301→  }
   302→   302→  submitRef.current = submitQuality
   303→   303→
   304→   304→  function advance() {
   305→   305→    // Flip back first, then slide to next card.
   306→   306→    setFlipped(false)
   307→   307→    window.setTimeout(() => {
   308→   308→      if (currentIndex + 1 >= sessionCards.length) {
   309→   309→        setMode('complete')
   310→   310→        fetchDue()
   311→   311→      } else {
   312→   312→        setDirection(1)
   313→   313→        setCurrentIndex((i) => i + 1)
   314→   314→      }
   315→   315→    }, 220)
   316→   316→  }
   317→   317→
   318→   318→  async function deleteCard() {
   319→   319→    const card = sessionCards[currentIndex]
   320→   320→    if (!card?.id || deleting) return
   321→   321→    setDeleting(true)
   322→   322→    try {
   323→   323→      const res = await fetch(`/api/flashcards/${card.id}`, { method: 'DELETE' })
   324→   324→      if (!res.ok) throw new Error('Не удалось удалить')
   325→   325→      toast.success('Карточка удалена')
   326→   326→      // Remove from session without counting as reviewed.
   327→   327→      setSessionCards((prev) => {
   328→   328→        const next = prev.filter((_, i) => i !== currentIndex)
   329→   329→        if (next.length === 0) {
   330→   330→          setMode('complete')
   331→   331→          return []
   332→   332→        }
   333→   333→        // Stay on same index if possible, else clamp.
   334→   334→        if (currentIndex >= next.length) {
   335→   335→          setCurrentIndex(next.length - 1)
   336→   336→        }
   337→   337→        return next
   338→   338→      })
   339→   339→      setFlipped(false)
   340→   340→    } catch (e) {
   341→   341→      toast.error('Не удалось удалить карточку')
   342→   342→    } finally {
   343→   343→      setDeleting(false)
   344→   344→    }
   345→   345→  }
   346→   346→
   347→   347→  function exitSession() {
   348→   348→    setMode('generate')
   349→   349→    setSessionCards([])
   350→   350→    setFlipped(false)
   351→   351→    setSessionSource(null)
   352→   352→    fetchDue()
   353→   353→  }
   354→   354→
   355→   355→  /* ------------------------- keyboard ------------------------------ */
   356→   356→
   357→   357→  useEffect(() => {
   358→   358→    if (mode !== 'review') return

   361→   361→      const tag = target?.tagName
   362→   362→      if (tag === 'INPUT' || tag === 'TEXTAREA' || target?.isContentEditable) return

   364→   364→        e.preventDefault()
   365→   365→        flipRef.current()
   366→   366→        return
   367→   367→      }
   368→   368→      if (e.key === '1') {
   369→   369→        e.preventDefault()
   370→   370→        submitRef.current('again')
   371→   371→      } else if (e.key === '2') {
   372→   372→        e.preventDefault()
   373→   373→        submitRef.current('hard')
   374→   374→      } else if (e.key === '3') {
   375→   375→        e.preventDefault()
   376→   376→        submitRef.current('good')
   377→   377→      } else if (e.key === '4') {
   378→   378→        e.preventDefault()
   379→   379→        submitRef.current('easy')
   380→   380→      }
   381→   381→    }
   382→   382→    window.addEventListener('keydown', onKey)
   383→   383→    return () => window.removeEventListener('keydown', onKey)
   384→   384→  }, [mode])
   385→   385→
   386→   386→  /* --------------------------- render ------------------------------ */

   398→   398→        onFlip={flip}
   399→   399→        onQuality={submitQuality}
   400→   400→        onDelete={deleteCard}
   401→   401→        onExit={exitSession}
   402→   402→      />
   403→   403→    )
   404→   404→  }

   409→   409→        reviewedCount={reviewedCount}
   410→   410→        totalCards={sessionCards.length || reviewedCount}
   411→   411→        xpGained={xpGained}
   412→   412→        onBack={exitSession}
   413→   413→      />
   414→   414→    )
   415→   415→  }
   416→   416→
   417→   417→  return (
   418→   418→    <PageSection className="py-8">
   419→   419→      <SectionHeader
   420→   420→        title="Флешкарты"
   421→   421→        subtitle="Запоминай надолго с интервальным повторением"
   422→   422→        icon={Layers}
   423→   423→        action={
   424→   424→          <Pill className="hidden sm:inline-flex border-primary/30 bg-primary/10 text-primary">
   425→   425→            <Zap className="h-3 w-3" />
   426→   426→            SM-2 алгоритм
   427→   427→          </Pill>
   428→   428→        }
   429→   429→      />
   430→   430→
   431→   431→      {/* Due-for-review banner */}
   432→   432→      <AnimatePresence>
   433→   433→        {!loadingDue && dueCards.length > 0 && (
   434→   434→          <motion.div
   435→   435→            initial={{ opacity: 0, y: -8 }}
   436→   436→            animate={{ opacity: 1, y: 0 }}
   437→   437→            exit={{ opacity: 0, y: -8 }}
   438→   438→            className="mb-6"
   439→   439→          >
   440→   440→            <GlassCard
   441→   441→              gradient
   442→   442→              hover={false}
   443→   443→              className="overflow-hidden border-primary/30"
   444→   444→            >
   445→   445→              <div className="relative flex flex-wrap items-center justify-between gap-4 p-5 sm:p-6">
   446→   446→                <div className="pointer-events-none absolute -right-10 -top-10 h-40 w-40 rounded-full bg-primary/20 blur-3xl" />
   447→   447→                <div className="relative flex items-center gap-4">
   448→   448→                  <div className="grid h-12 w-12 shrink-0 place-items-center rounded-xl bg-gradient-to-br from-violet-500 to-fuchsia-500 text-white shadow-lg shadow-fuchsia-500/30">
   449→   449→                    <Repeat className="h-6 w-6" />
   450→   450→                  </div>
   451→   451→                  <div>
   452→   452→                    <p className="font-semibold leading-tight">
   453→   453→                      У тебя{' '}
   454→   454→                      <span className="text-gradient">{dueCards.length}</span>{' '}
   455→   455→                      {pluralizeCards(dueCards.length)} на повторение
   456→   456→                    </p>
   457→   457→                    <p className="mt-0.5 text-sm text-muted-foreground">
   458→   458→                      Самое время освежить память — это занимает пару минут
   459→   459→                    </p>
   460→   460→                  </div>
   461→   461→                </div>
   462→   462→                <GradientButton onClick={startDueReview} className="shrink-0">
   463→   463→                  <RotateCcw className="h-4 w-4" />
   464→   464→                  Повторить
   465→   465→                </GradientButton>
   466→   466→              </div>
   467→   467→            </GlassCard>
   468→   468→          </motion.div>
   469→   469→        )}
   470→   470→      </AnimatePresence>
   471→   471→
   472→   472→      {/* Generate form */}
   473→   473→      <GlassCard className="mb-6 overflow-hidden" hover={false}>
   474→   474→        <div className="relative border-b border-border/60 bg-gradient-to-br from-violet-500/5 via-fuchsia-500/5 to-transparent p-5 sm:p-6">
   475→   475→          <div className="pointer-events-none absolute inset-0 bg-grid opacity-10" />
   476→   476→          <div className="relative">
   477→   477→            <label
   478→   478→              htmlFor="flash-topic"
   479→   479→              className="mb-2 block text-sm font-medium"
   480→   480→            >
   481→   481→              Какую тему закрепляем?
   482→   482→            </label>
   483→   483→            <div className="flex flex-col gap-3 sm:flex-row">
   484→   484→              <input
