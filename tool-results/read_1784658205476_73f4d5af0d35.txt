    81→function difficultyMeta(d: string) {
    82→  const diff = (d || '').toLowerCase()
    83→  if (
    84→    diff.startsWith('easy') ||
    85→    diff.startsWith('лёг') ||
    86→    diff.startsWith('лег') ||
    87→    diff.startsWith('прост')
    88→  ) {
    89→    return {
    90→      label: 'Легко',
    91→      cls: 'border-emerald-500/40 bg-emerald-500/10 text-emerald-300',
    92→      dot: 'bg-emerald-400',
    93→    }
    94→  }
    95→  if (
    96→    diff.startsWith('hard') ||
    97→    diff.startsWith('слож') ||
    98→    diff.startsWith('тяж') ||
    99→    diff.startsWith('прод')
   100→  ) {
   101→    return {
   102→      label: 'Сложно',
   103→      cls: 'border-rose-500/40 bg-rose-500/10 text-rose-300',
   104→      dot: 'bg-rose-400',
   105→    }
   106→  }
   107→  return {
   108→    label: 'Средне',
   109→    cls: 'border-amber-500/40 bg-amber-500/10 text-amber-300',
   110→    dot: 'bg-amber-400',
   111→  }
   112→}

   114→function formatTime(sec: number) {
   115→  const m = Math.floor(sec / 60)
   116→  const s = sec % 60
   117→  return `${m}:${s.toString().padStart(2, '0')}`
   118→}

   122→export function QuizView() {
   123→  const { activeSubject } = useNav()
   124→  const [phase, setPhase] = useState<Phase>('setup')
   125→  const [topic, setTopic] = useState('')
   126→  const [subject, setSubject] = useState<string>(activeSubject || 'programming')
   127→  const [count, setCount] = useState<number>(5)
   128→  const [level, setLevel] = useState<string>('Любой')
   129→  const [questions, setQuestions] = useState<QuizQuestion[]>([])
   130→  const [answers, setAnswers] = useState<(number | null | undefined)[]>([])
   131→  const [currentIdx, setCurrentIdx] = useState(0)
   132→  const [revealed, setRevealed] = useState(false)
   133→  const [elapsed, setElapsed] = useState(0)
   134→  const [quizResult, setQuizResult] = useState<QuizResult | null>(null)
   135→  const [submitting, setSubmitting] = useState(false)
   136→
   137→  // Prefill subject from nav if available
   138→  useEffect(() => {
   139→    if (activeSubject) setSubject(activeSubject)
   140→  }, [activeSubject])
   141→
   142→  // Per-question elapsed timer (stops once revealed)
   143→  useEffect(() => {
   144→    setElapsed(0)
   145→  }, [currentIdx])
   146→  useEffect(() => {
   147→    if (phase !== 'quiz' || revealed) return
   148→    const id = setInterval(() => setElapsed((e) => e + 1), 1000)
   149→    return () => clearInterval(id)
   150→  }, [phase, currentIdx, revealed])
   151→
   152→  const suggestedTopics = useMemo(() => {
   153→    const subj = getSubject(subject)
   154→    return subj?.topics?.slice(0, 6) ?? FALLBACK_TOPICS
   155→  }, [subject])
   156→
   157→  const currentScore = useMemo(() => {
   158→    return questions.reduce((acc, q, i) => {
   159→      const isRevealed = i < currentIdx || (i === currentIdx && revealed)
   160→      if (isRevealed && answers[i] === q.correctIndex) return acc + 1
   161→      return acc
   162→    }, 0)
   163→  }, [questions, currentIdx, revealed, answers])
   164→
   165→  const addressedCount = useMemo(
   166→    () => answers.filter((a) => a !== undefined).length,
   167→    [answers]
   168→  )
   169→
   170→  async function startQuiz(opts?: {
   171→    topic?: string
   172→    subject?: string
   173→    count?: number
   174→    level?: string
   175→  }) {
   176→    const t = (opts?.topic ?? topic).trim()
   177→    const s = opts?.subject ?? subject
   178→    const c = opts?.count ?? count
   179→    const l = opts?.level ?? level
   180→    if (!t) {
   181→      toast.error('Укажите тему квиза')
   182→      return
   183→    }
   184→    setTopic(t)
   185→    setSubject(s)
   186→    setCount(c)
   187→    setLevel(l)
   188→    setPhase('loading')
   189→    setQuizResult(null)
   190→    try {
   191→      const res = await fetch('/api/quiz', {
   192→        method: 'POST',
   193→        headers: { 'Content-Type': 'application/json' },
   194→        body: JSON.stringify({
   195→          topic: t,
   196→          subject: s,
   197→          count: c,
   198→          level: l === 'Любой' ? undefined : l,
   199→        }),
   200→      })
   201→      const data = await res.json()
   202→      if (!res.ok) throw new Error(data.error)
   203→      if (!data.questions?.length) throw new Error('Нет вопросов')
   204→      setQuestions(data.questions)
   205→      setAnswers(Array(data.questions.length).fill(undefined))
   206→      setCurrentIdx(0)
   207→      setRevealed(false)
   208→      setElapsed(0)
   209→      setPhase('quiz')
   210→      toast.success('Квиз готов! Удачи 🎯')
   211→    } catch {
   212→      toast.error('Не удалось создать квиз. Попробуйте ещё раз.')
   213→      setPhase('setup')
   214→    }
   215→  }
   216→
   217→  function selectOption(idx: number) {
   218→    if (revealed) return
   219→    const next = [...answers]
   220→    next[currentIdx] = idx
   221→    setAnswers(next)
   222→  }
   223→
   224→  function handleCheck() {
   225→    if (answers[currentIdx] === undefined || answers[currentIdx] === null) {
   226→      toast.error('Выбери вариант ответа')
   227→      return
   228→    }
   229→    setRevealed(true)
   230→  }
   231→
   232→  function handleSkip() {
   233→    if (revealed) return
   234→    const next = [...answers]
   235→    next[currentIdx] = null
   236→    setAnswers(next)
   237→    setRevealed(true)
   238→  }
   239→
   240→  async function handleNext() {
   241→    if (currentIdx + 1 >= questions.length) {
   242→      await finishQuiz()
   243→    } else {
   244→      setCurrentIdx(currentIdx + 1)
   245→      setRevealed(false)
   246→    }
   247→  }
   248→
   249→  async function finishQuiz() {
   250→    const correct = questions.reduce(
   251→      (acc, q, i) => acc + (answers[i] === q.correctIndex ? 1 : 0),
   252→      0
   253→    )
   254→    setPhase('results')
   255→    setSubmitting(true)
   256→    try {
   257→      const res = await fetch('/api/quiz', {
   258→        method: 'PUT',
   259→        headers: { 'Content-Type': 'application/json' },
   260→        body: JSON.stringify({
   261→          subject,
   262→          topic,
   263→          total: questions.length,
   264→          correct,
   265→        }),
   266→      })
   267→      const data = await res.json()
   268→      if (!res.ok) throw new Error(data.error)
   269→      setQuizResult(data)
   270→      useUser.setState({ xp: data.xp, level: data.level })
   271→      if (data.xpGain > 0) {
   272→        toast.success(`+${data.xpGain} XP`, { description: 'Прогресс сохранён' })
   273→      }
   274→    } catch {
   275→      toast.error('Не удалось сохранить прогресс')
   276→    } finally {
   277→      setSubmitting(false)
   278→    }
   279→  }
   280→
   281→  function resetToSetup(clearTopic = true) {
   282→    setPhase('setup')
   283→    setQuestions([])
   284→    setAnswers([])
   285→    setCurrentIdx(0)
   286→    setRevealed(false)
   287→    setQuizResult(null)
   288→    if (clearTopic) setTopic('')
   289→  }
   290→
   291→  return (
   292→    <PageSection className="py-8">
   293→      <SectionHeader
   294→        title="Квиз-арена"
   295→        subtitle="Адаптивный квиз по любой теме — проверь себя и заработай XP"
   296→        icon={Trophy}
   297→      />
   298→
   299→      <AnimatePresence mode="wait">
   300→        {phase === 'setup' && (
   301→          <motion.div
   302→            key="setup"
   303→            initial={{ opacity: 0, y: 12 }}
   304→            animate={{ opacity: 1, y: 0 }}
   305→            exit={{ opacity: 0, y: -12 }}
   306→            transition={{ duration: 0.3 }}
   307→          >
   308→            <SetupPhase
   309→              topic={topic}
   310→              setTopic={setTopic}
   311→              subject={subject}
   312→              setSubject={setSubject}
   313→              count={count}
   314→              setCount={setCount}
   315→              level={level}
   316→              setLevel={setLevel}
   317→              suggestedTopics={suggestedTopics}
   318→              onStart={() => startQuiz()}
   319→            />
   320→          </motion.div>
   321→        )}
   322→
   323→        {phase === 'loading' && (
   324→          <motion.div
   325→            key="loading"
   326→            initial={{ opacity: 0 }}
   327→            animate={{ opacity: 1 }}
   328→            exit={{ opacity: 0 }}
   329→          >
   330→            <QuizSkeleton />
   331→          </motion.div>
   332→        )}
   333→
   334→        {phase === 'quiz' && questions.length > 0 && (
   335→          <motion.div
   336→            key="quiz"
   337→            initial={{ opacity: 0 }}
   338→            animate={{ opacity: 1 }}
   339→            exit={{ opacity: 0 }}
   340→          >
   341→            <QuizPhase
   342→              question={questions[currentIdx]}
   343→              questionNumber={currentIdx + 1}
   344→              total={questions.length}
   345→              selected={answers[currentIdx]}
   346→              revealed={revealed}
   347→              score={currentScore}
   348→              elapsed={elapsed}
   349→              progressValue={(addressedCount / questions.length) * 100}
   350→              onSelect={selectOption}
   351→              onCheck={handleCheck}
   352→              onSkip={handleSkip}
   353→              onNext={handleNext}
   354→              onExit={() => resetToSetup(true)}
   355→              isLast={currentIdx + 1 >= questions.length}
   356→            />
   357→          </motion.div>
   358→        )}
   359→
   360→        {phase === 'results' && (
   361→          <motion.div
   362→            key="results"
   363→            initial={{ opacity: 0 }}
   364→            animate={{ opacity: 1 }}
   365→            exit={{ opacity: 0 }}
   366→          >
   367→            <ResultsPhase
   368→              questions={questions}
   369→              answers={answers}
   370→              topic={topic}
   371→              subject={subject}
   372→              result={quizResult}
   373→              submitting={submitting}
   374→              onNewQuiz={() => resetToSetup(true)}
   375→              onRetryTopic={() => startQuiz({ topic })}
   376→            />
   377→          </motion.div>
   378→        )}
   379→      </AnimatePresence>
   380→    </PageSection>
   381→  )
   382→}

   596→function QuizPhase({

  1117→function Confetti({ run }: { run: boolean }) {
  1118→  const pieces = useMemo(
  1119→    () =>
  1120→      Array.from({ length: 70 }, (_, i) => ({
  1121→        id: i,
  1122→        left: Math.random() * 100,
  1123→        delay: Math.random() * 0.5,

  1125→        rotate: Math.random() * 720 - 360,
  1126→        color: [
  1127→          'bg-violet-400',
  1128→          'bg-fuchsia-400',
  1129→          'bg-pink-400',
  1130→          'bg-amber-300',
  1131→          'bg-emerald-300',
  1132→          'bg-rose-300',
  1133→        ][i % 6],
  1134→        size: 6 + Math.random() * 8,
  1135→        rounded: Math.random() > 0.5,
  1136→      })),
  1137→    []

  1139→  if (!run) return null
  1140→  return (
  1141→    <div

  1143→      className="pointer-events-none absolute inset-0 z-10 overflow-hidden"
  1144→    >
  1145→      {pieces.map((p) => (

  1147→          key={p.id}
  1148→          className={cn('absolute top-0', p.color, p.rounded ? 'rounded-full' : 'rounded-sm')}
  1149→          style={{ left: `${p.left}%`, width: p.size, height: p.size }}
  1150→          initial={{ y: -40, opacity: 0, rotate: 0 }}
  1151→          animate={{ y: ['0vh', '110vh'], opacity: [0, 1, 1, 0], rotate: p.rotate }}
