   109→   109→  // ---------- data fns ----------
   110→   110→  async function loadSessions() {
   111→   111→    setLoadingSessions(true)
   112→   112→    try {
   113→   113→      const res = await fetch('/api/chat', { cache: 'no-store' })
   114→   114→      const data = await res.json()

   116→   116→        setSessions(data.sessions as ChatSessionMeta[])

   125→   125→  async function loadThread(sid: string) {
   126→   126→    setLoadingThread(true)
   127→   127→    try {
   128→   128→      const res = await fetch(`/api/chat/${sid}`, { cache: 'no-store' })
   129→   129→      const data = await res.json()
   130→   130→      if (!res.ok) throw new Error(data?.error || 'Не удалось загрузить диалог')
   131→   131→      const apiMsgs = (data.session?.messages ?? []) as ApiMessage[]
   132→   132→      const msgs: ChatMessage[] = apiMsgs.map((m, i) => ({
   133→   133→        id: `${sid}-${i}`,
   134→   134→        role: m.role,
   135→   135→        content: m.content,
   136→   136→      }))
   137→   137→      setMessages(msgs)
   138→   138→      if (data.session?.subject && data.session.subject !== 'general') {
   139→   139→        setSubject(data.session.subject)
   140→   140→      }
   141→   141→      setSessionId(sid)
   142→   142→      sessionStorage.setItem(SESSION_KEY, sid)
   143→   143→    } catch (e) {
   144→   144→      const msg = e instanceof Error ? e.message : 'Не удалось загрузить диалог'
   145→   145→      toast.error(msg)
   146→   146→      setSessionId(null)
   147→   147→      sessionStorage.removeItem(SESSION_KEY)
   148→   148→    } finally {
   149→   149→      setLoadingThread(false)
   150→   150→      setListOpen(false)
   151→   151→    }
   152→   152→  }

   154→   154→  async function deleteSession(sid: string, e?: React.MouseEvent) {
   155→   155→    e?.stopPropagation()
   156→   156→    try {
   157→   157→      const res = await fetch(`/api/chat/${sid}`, { method: 'DELETE' })
   158→   158→      if (!res.ok) throw new Error('Не удалось удалить')
   159→   159→      setSessions((s) => s.filter((x) => x.id !== sid))
   160→   160→      if (sessionId === sid) {
   161→   161→        setSessionId(null)
   162→   162→        setMessages([])
   163→   163→        sessionStorage.removeItem(SESSION_KEY)
   164→   164→      }
   165→   165→      toast.success('Диалог удалён')
   166→   166→    } catch {
   167→   167→      toast.error('Не удалось удалить диалог')
   168→   168→    }
   169→   169→  }

   179→   179→  // ---------- send ----------
   180→   180→  async function send(text?: string, overrideSubject?: string) {
   181→   181→    const content = (text ?? input).trim()
   182→   182→    if (!content || sending) return
   183→   183→    setInput('')
   184→   184→
   185→   185→    const usedSubject = overrideSubject ?? subject
   186→   186→
   187→   187→    const userMsg: ChatMessage = {
   188→   188→      id: `u-${Date.now()}`,
   189→   189→      role: 'user',
   190→   190→      content,
   191→   191→    }
   192→   192→    const prevMessages = messages
   193→   193→    const nextMessages = [...prevMessages, userMsg]
   194→   194→    setMessages(nextMessages)
   195→   195→    setSending(true)
   196→   196→
   197→   197→    try {
   198→   198→      const history = prevMessages.map(
   199→   199→        (m): ApiMessage => ({ role: m.role, content: m.content })
   200→   200→      )
   201→   201→      const res = await fetch('/api/chat', {
   202→   202→        method: 'POST',
   203→   203→        headers: { 'Content-Type': 'application/json' },
   204→   204→        body: JSON.stringify({
   205→   205→          message: content,
   206→   206→          subject: usedSubject,
   207→   207→          sessionId,
   208→   208→          history,
   209→   209→        }),
   210→   210→      })
   211→   211→      const data = await res.json()
   212→   212→      if (!res.ok) throw new Error(data?.error || 'Не удалось получить ответ')
   213→   213→
   214→   214→      const aiMsg: ChatMessage = {
   215→   215→        id: `a-${Date.now()}`,
   216→   216→        role: 'assistant',
   217→   217→        content: data.reply as string,
   218→   218→      }
   219→   219→      setMessages((m) => [...m, aiMsg])
   220→   220→
   221→   221→      const newSid: string | undefined = data.sessionId
   222→   222→      if (newSid) {
   223→   223→        setSessionId(newSid)
   224→   224→        sessionStorage.setItem(SESSION_KEY, newSid)
   225→   225→      }
   226→   226→      if (typeof data.xp === 'number' && typeof data.level === 'number') {
   227→   227→        useUser.setState({ xp: data.xp, level: data.level })
   228→   228→      }
   229→   229→      void loadSessions()
   230→   230→    } catch (e) {
   231→   231→      const msg = e instanceof Error ? e.message : 'Не удалось получить ответ'
   232→   232→      toast.error(msg)
   233→   233→    } finally {
   234→   234→      setSending(false)
   235→   235→    }
   236→   236→  }

   720→   720→function formatDate(iso: string): string {
   721→   721→  try {
   722→   722→    const d = new Date(iso)
   723→   723→    const now = new Date()
   724→   724→    const diffMs = now.getTime() - d.getTime()
   725→   725→    const diffMin = Math.floor(diffMs / 60000)
   726→   726→    if (diffMin < 1) return 'только что'
   727→   727→    if (diffMin < 60) return `${diffMin} мин назад`
   728→   728→    const diffH = Math.floor(diffMin / 60)
   729→   729→    if (diffH < 24) return `${diffH} ч назад`
   730→   730→    const diffD = Math.floor(diffH / 24)
   731→   731→    if (diffD < 7) return `${diffD} д назад`
   732→   732→    return d.toLocaleDateString('ru-RU', { day: 'numeric', month: 'short' })
   733→   733→  } catch {
   734→   734→    return ''
   735→   735→  }
   736→   736→}
