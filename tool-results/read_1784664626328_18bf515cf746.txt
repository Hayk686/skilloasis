   191→interface DaySeriesItem {

   194→  count: number
   195→  isToday: boolean

   197→
   198→function build7DaySeries(byDay: Record<string, number>): DaySeriesItem[] {
   199→  const days: DaySeriesItem[] = []
   200→  const today = new Date()
   201→  for (let i = 6; i >= 0; i--) {
   202→    const d = new Date(today)
   203→    d.setDate(today.getDate() - i)
   204→    const key = d.toISOString().slice(0, 10)
   205→    const weekday = d.toLocaleDateString('ru-RU', { weekday: 'short' })
   206→    const isToday = i === 0
   207→    days.push({
   208→      key,
   209→      label: weekday.charAt(0).toUpperCase() + weekday.slice(1),
   210→      count: byDay[key] || 0,
   211→      isToday,
   212→    })
   213→  }
   214→  return days
   215→}

   263→function XpRing({
   264→  pct,
   265→  level,
   266→}: {
   267→  pct: number
   268→  level: number
   269→}) {

   272→  const dash = (Math.max(0, Math.min(100, pct)) / 100) * c
   273→  return (
   274→    <div className="relative h-32 w-32 shrink-0">
   275→      <svg className="h-full w-full -rotate-90" viewBox="0 0 128 128">
   276→        <defs>
   277→          <linearGradient id="xpRingGrad" x1="0%" y1="0%" x2="100%" y2="100%">
   278→            <stop offset="0%" stopColor="oklch(0.7 0.25 315)" />
   279→            <stop offset="55%" stopColor="oklch(0.72 0.25 350)" />
   280→            <stop offset="100%" stopColor="oklch(0.78 0.22 55)" />
   281→          </linearGradient>
   282→        </defs>
   283→        <circle
   284→          cx="64"
   285→          cy="64"
   286→          r={r}
   287→          fill="none"
   288→          stroke="currentColor"
   289→          strokeWidth="9"
   290→          className="text-muted/40"
   291→        />
   292→        <motion.circle
   293→          cx="64"
   294→          cy="64"
   295→          r={r}
   296→          fill="none"
   297→          stroke="url(#xpRingGrad)"
   298→          strokeWidth="9"
   299→          strokeLinecap="round"
   300→          strokeDasharray={`${dash} ${c - dash}`}
   301→          initial={{ strokeDasharray: `0 ${c}` }}
   302→          animate={{ strokeDasharray: `${dash} ${c - dash}` }}
   303→          transition={{ duration: 1.1, ease: 'easeOut' }}
   304→          style={{ filter: 'drop-shadow(0 0 8px oklch(0.7 0.25 315 / 0.55))' }}

   315→    </div>
   316→  )

   319→/* ============================================================ *
   320→ * Confetti burst
   321→ * ============================================================ */
   322→
   323→function ConfettiBurst({ burst }: { burst: number }) {
   324→  const particles = useMemo(() => {
   325→    if (burst === 0) return []
   326→    return Array.from({ length: 28 }, (_, i) => {
   327→      const angle = (i / 28) * Math.PI * 2 + Math.random() * 0.4
   328→      const distance = 90 + Math.random() * 110
   329→      return {
   330→        id: `${burst}-${i}`,
   331→        x: Math.cos(angle) * distance,
   332→        y: Math.sin(angle) * distance,
   333→        delay: Math.random() * 0.12,
   334→        emoji: ['🎉', '✨', '⭐', '🔥', '💫', '🌟'][i % 6],
   335→        rotate: (Math.random() - 0.5) * 540,
   336→      }
   337→    })
   338→  }, [burst])
   339→
   340→  return (
   341→    <div className="pointer-events-none absolute inset-0 flex items-center justify-center overflow-visible">
   342→      <AnimatePresence>
   343→        {particles.map((p) => (
   344→          <motion.div
   345→            key={p.id}
   346→            className="absolute text-lg"
   347→            initial={{ x: 0, y: 0, opacity: 1, scale: 0, rotate: 0 }}
   348→            animate={{
   349→              x: p.x,
   350→              y: p.y,
   351→              opacity: 0,
   352→              scale: 1.1,
   353→              rotate: p.rotate,
   354→            }}
   355→            transition={{ duration: 1.5, delay: p.delay, ease: 'easeOut' }}
   356→          >
   357→            {p.emoji}
   358→          </motion.div>
   359→        ))}
   360→      </AnimatePresence>
   361→    </div>
   362→  )
   363→}

   453→
   454→function ChartTooltip({ active, payload, label }: TooltipProps<number, string>) {
   455→  if (!active || !payload || payload.length === 0) return null
   456→  const v = Number(payload[0]?.value ?? 0)
   457→  return (
   458→    <div className="rounded-xl border border-border/70 bg-card/95 px-3 py-2 text-xs shadow-xl backdrop-blur-md">
   459→      <div className="font-semibold text-foreground">{label}</div>
   460→      <div className="mt-0.5 text-muted-foreground">
   461→        {v === 0 ? 'нет активности' : `${v} ${pluralActivity(v)}`}
   462→      </div>
   463→    </div>
   464→  )
   465→}
   466→
   467→function pluralActivity(n: number) {
   468→  const mod10 = n % 10
   469→  const mod100 = n % 100
   470→  if (mod10 === 1 && mod100 !== 11) return 'действие'
   471→  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 10 || mod100 >= 20))
   472→    return 'действия'
   473→  return 'действий'
   474→}

   496→  const [savingName, setSavingName] = useState(false)

   499→  // daily challenge

   501→  const [submittingDaily, setSubmittingDaily] = useState(false)
   502→  const [hintOpen, setHintOpen] = useState(false)
   503→  const [confettiBurst, setConfettiBurst] = useState(0)
   504→  const [shareOpen, setShareOpen] = useState(false)

   506→  /* ---------- fetch all in parallel ---------- */
   507→  useEffect(() => {
   508→    let cancelled = false
   509→    async function load() {
   510→      setLoading(true)
   511→      try {
   512→        const [uRes, pRes, dRes] = await Promise.all([
   513→          fetch('/api/user', { cache: 'no-store' }),

   541→      } finally {
   542→        if (!cancelled) setLoading(false)
   543→      }
   544→    }

   546→    return () => {
   547→      cancelled = true

   587→      setEditingName(false)

  1457→function pluralDays(n: number) {
  1458→  const mod10 = n % 10
  1459→  const mod100 = n % 100
  1460→  if (mod10 === 1 && mod100 !== 11) return 'день'
