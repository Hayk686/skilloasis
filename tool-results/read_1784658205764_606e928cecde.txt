   112→
   113→export function FlashcardsView() {
   114→  const { activeSubject } = useNav()
   115→  const user = useUser()
   116→
   117→  // Top-level mode
   118→  const [mode, setMode] = useState<Mode>('generate')
   119→
   120→  // Generate form state
   121→  const [topic, setTopic] = useState('')
   122→  const [subject, setSubjectLocal] = useState<string>(
   123→    activeSubject && SUBJECTS.some((s) => s.id === activeSubject)
   124→      ? activeSubject
   125→      : 'programming'
   126→  )
   127→  const [count, setCount] = useState<number>(8)
   128→  const [generating, setGenerating] = useState(false)
   129→
   130→  // Due cards for the banner
   131→  const [dueCards, setDueCards] = useState<FlashCard[]>([])
   132→  const [loadingDue, setLoadingDue] = useState(true)
   133→
   134→  // Review session state
   135→  const [sessionCards, setSessionCards] = useState<FlashCard[]>([])
   136→  const [sessionSource, setSessionSource] = useState<SessionSource | null>(null)
   137→  const [currentIndex, setCurrentIndex] = useState(0)
   138→  const [flipped, setFlipped] = useState(false)
   139→  const [direction, setDirection] = useState(1)
   140→  const [reviewedCount, setReviewedCount] = useState(0)
   141→  const [xpGained, setXpGained] = useState(0)
   142→  const [submitting, setSubmitting] = useState(false)
   143→  const [deleting, setDeleting] = useState(false)
   144→
   145→  // Track latest submit to avoid stale closures in keyboard handler
   146→  const submitRef = useRef<(q: Quality) => void>(() => {})
   147→  const flipRef = useRef<() => void>(() => {})
   148→
   149→  /* ----------------------------- effects --------------------------- */
   150→
   151→  useEffect(() => {
   152→    if (activeSubject && SUBJECTS.some((s) => s.id === activeSubject)) {
   153→      setSubjectLocal(activeSubject)
   154→    }
   155→  }, [activeSubject])
   156→
   157→  const fetchDue = useCallback(async () => {
   158→    setLoadingDue(true)
   159→    try {
   160→      const res = await fetch('/api/flashcards?limit=50', { cache: 'no-store' })
   161→      const data = await res.json()
   162→      setDueCards(Array.isArray(data.cards) ? data.cards : [])
   163→    } catch {
   164→      setDueCards([])
   165→    } finally {
   166→      setLoadingDue(false)
   167→    }
   168→  }, [])
   169→
   170→  useEffect(() => {
   171→    fetchDue()
   172→  }, [fetchDue])
   173→
   174→  /* --------------------------- generate ---------------------------- */
   175→
   176→  async function handleGenerate(givenTopic?: string) {
   177→    const t = (givenTopic ?? topic).trim()
   178→    if (!t) {
   179→      toast.error('Введи тему для карточек')
   180→      return
   181→    }
   182→    setTopic(t)
   183→    setGenerating(true)
   184→    try {
   185→      const res = await fetch('/api/flashcards', {
   186→        method: 'POST',
   187→        headers: { 'Content-Type': 'application/json' },
   188→        body: JSON.stringify({ topic: t, subject, count }),
   189→      })
   190→      const data = await res.json()
   191→      if (!res.ok) throw new Error(data.error || 'Ошибка генерации')
   192→      if (!data.cards?.length) throw new Error('Карты не сгенерированы')
   193→
   194→      useUser.setState({
   195→        xp: typeof data.xp === 'number' ? data.xp : user.xp,
   196→        level: typeof data.level === 'number' ? data.level : user.level,
   197→      })
   198→
   199→      toast.success(`Создано ${data.cards.length} карточек!`, {
   200→        description: 'Начинаем сессию повторения',
   201→      })
   202→
   203→      // The POST returns cards without ids. Refetch due list to get the
   204→      // persisted versions (newly created cards default dueAt = now).
   205→      const dueRes = await fetch('/api/flashcards?limit=50', { cache: 'no-store' })
   206→      const dueData = await dueRes.json()
   207→      const all: FlashCard[] = Array.isArray(dueData.cards) ? dueData.cards : []
   208→      setDueCards(all)
   209→
   210→      // Match persisted cards by front text to ensure ids are present.
   211→      const generatedFronts = new Set(
   212→        (data.cards as FlashCard[]).map((c) => c.front.trim())
   213→      )
   214→      let session: FlashCard[] = all.filter((c) =>
   215→        generatedFronts.has((c.front || '').trim())
   216→      )
   217→      // Fallback: if matching failed for some reason, use the generated
   218→      // payloads directly (PATCH will be skipped for cards without ids).
   219→      if (session.length !== data.cards.length) {
   220→        session = data.cards as FlashCard[]
   221→      }
   222→
   223→      startSession(session, 'generated')
   224→    } catch (e) {
   225→      console.error('[flashcards] generate', e)
   226→      toast.error('Не удалось создать карты. Попробуй ещё раз.')
   227→    } finally {
   228→      setGenerating(false)
   229→    }
   230→  }
   231→
   232→  /* ---------------------------- session ---------------------------- */
   233→
   234→  function startSession(cards: FlashCard[], source: SessionSource) {
   235→    if (!cards.length) {
   236→      toast.error('Нет карточек для повторения')
   237→      return
   238→    }
   239→    setSessionCards(cards)
   240→    setSessionSource(source)
   241→    setCurrentIndex(0)
   242→    setFlipped(false)
   243→    setDirection(1)
   244→    setReviewedCount(0)
   245→    setXpGained(0)
   246→    setSubmitting(false)
   247→    setDeleting(false)
   248→    setMode('review')
   249→  }
   250→
   251→  function startDueReview() {
   252→    if (!dueCards.length) return
   253→    startSession(dueCards, 'due')
   254→  }
   255→
   256→  function flip() {
   257→    setFlipped((f) => !f)
   258→  }
   259→  flipRef.current = flip
   260→
   261→  async function submitQuality(quality: Quality) {
   262→    if (submitting || deleting) return
   263→    const card = sessionCards[currentIndex]
   264→    if (!card) return
   265→
   266→    // No id (e.g. generated fallback) — just advance.
   267→    if (!card.id) {
   268→      setReviewedCount((v) => v + 1)
   269→      advance()
   270→      return
   271→    }
   272→
   273→    setSubmitting(true)
   274→    try {
   275→      const res = await fetch(`/api/flashcards/${card.id}`, {
   276→        method: 'PATCH',
   277→        headers: { 'Content-Type': 'application/json' },
   278→        body: JSON.stringify({ quality }),
   279→      })
   280→      const data = await res.json()
   281→      if (!res.ok) throw new Error(data.error || 'Ошибка')
   282→
   283→      useUser.setState({
   284→        xp: typeof data.xp === 'number' ? data.xp : user.xp,
   285→        level: typeof data.level === 'number' ? data.level : user.level,
   286→      })
   287→
   288→      let gain = 0
   289→      if (quality === 'easy') gain = 5
   290→      else if (quality === 'good') gain = 3
   291→      setXpGained((v) => v + gain)
   292→      setReviewedCount((v) => v + 1)
   293→
   294→      advance()
   295→    } catch (e) {
   296→      console.error('[flashcards] submit', e)
   297→      toast.error('Не удалось сохранить ответ')
   298→    } finally {
   299→      setSubmitting(false)
   300→    }
   301→  }
   302→  submitRef.current = submitQuality
   303→
   304→  function advance() {
   305→    // Flip back first, then slide to next card.
   306→    setFlipped(false)
   307→    window.setTimeout(() => {
   308→      if (currentIndex + 1 >= sessionCards.length) {
   309→        setMode('complete')
   310→        fetchDue()
   311→      } else {
   312→        setDirection(1)
   313→        setCurrentIndex((i) => i + 1)
   314→      }
   315→    }, 220)
   316→  }
   317→
   318→  async function deleteCard() {
   319→    const card = sessionCards[currentIndex]
   320→    if (!card?.id || deleting) return
   321→    setDeleting(true)
   322→    try {
   323→      const res = await fetch(`/api/flashcards/${card.id}`, { method: 'DELETE' })
   324→      if (!res.ok) throw new Error('Не удалось удалить')
   325→      toast.success('Карточка удалена')
   326→      // Remove from session without counting as reviewed.
   327→      setSessionCards((prev) => {
   328→        const next = prev.filter((_, i) => i !== currentIndex)
   329→        if (next.length === 0) {
   330→          setMode('complete')
   331→          return []
   332→        }
   333→        // Stay on same index if possible, else clamp.
   334→        if (currentIndex >= next.length) {
   335→          setCurrentIndex(next.length - 1)
   336→        }
   337→        return next
   338→      })
   339→      setFlipped(false)
   340→    } catch (e) {
   341→      toast.error('Не удалось удалить карточку')
   342→    } finally {
   343→      setDeleting(false)
   344→    }
   345→  }
   346→
   347→  function exitSession() {
   348→    setMode('generate')
   349→    setSessionCards([])
   350→    setFlipped(false)
   351→    setSessionSource(null)
   352→    fetchDue()
   353→  }
   354→
   355→  /* ------------------------- keyboard ------------------------------ */
   356→
   357→  useEffect(() => {
   358→    if (mode !== 'review') return
   359→    function onKey(e: KeyboardEvent) {
   360→      const target = e.target as HTMLElement | null
   361→      const tag = target?.tagName
   362→      if (tag === 'INPUT' || tag === 'TEXTAREA' || target?.isContentEditable) return
   363→      if (e.code === 'Space') {
   364→        e.preventDefault()
   365→        flipRef.current()
   366→        return
   367→      }
   368→      if (e.key === '1') {
   369→        e.preventDefault()
   370→        submitRef.current('again')
   371→      } else if (e.key === '2') {
   372→        e.preventDefault()
   373→        submitRef.current('hard')
   374→      } else if (e.key === '3') {
   375→        e.preventDefault()
   376→        submitRef.current('good')
   377→      } else if (e.key === '4') {
   378→        e.preventDefault()
   379→        submitRef.current('easy')
   380→      }
   381→    }
   382→    window.addEventListener('keydown', onKey)
   383→    return () => window.removeEventListener('keydown', onKey)
   384→  }, [mode])
   385→
   386→  /* --------------------------- render ------------------------------ */
   387→
   388→  if (mode === 'review') {
   389→    return (
   390→      <ReviewSession
   391→        cards={sessionCards}
   392→        index={currentIndex}
   393→        flipped={flipped}
   394→        direction={direction}
   395→        source={sessionSource}
   396→        submitting={submitting}
   397→        deleting={deleting}
   398→        onFlip={flip}
   399→        onQuality={submitQuality}
   400→        onDelete={deleteCard}
   401→        onExit={exitSession}
   402→      />
   403→    )
   404→  }
   405→
   406→  if (mode === 'complete') {
   407→    return (
   408→      <CompleteScreen
   409→        reviewedCount={reviewedCount}
   410→        totalCards={sessionCards.length || reviewedCount}
   411→        xpGained={xpGained}
   412→        onBack={exitSession}
   413→      />
   414→    )
   415→  }
   416→
   417→  return (
   418→    <PageSection className="py-8">
   419→      <SectionHeader
   420→        title="Флешкарты"
   421→        subtitle="Запоминай надолго с интервальным повторением"
   422→        icon={Layers}
   423→        action={
   424→          <Pill className="hidden sm:inline-flex border-primary/30 bg-primary/10 text-primary">
   425→            <Zap className="h-3 w-3" />
   426→            SM-2 алгоритм
   427→          </Pill>
   428→        }
   429→      />
   430→
   431→      {/* Due-for-review banner */}
   432→      <AnimatePresence>
   433→        {!loadingDue && dueCards.length > 0 && (
   434→          <motion.div
   435→            initial={{ opacity: 0, y: -8 }}
   436→            animate={{ opacity: 1, y: 0 }}
   437→            exit={{ opacity: 0, y: -8 }}
   438→            className="mb-6"
   439→          >
   440→            <GlassCard
   441→              gradient
   442→              hover={false}
   443→              className="overflow-hidden border-primary/30"
   444→            >
   445→              <div className="relative flex flex-wrap items-center justify-between gap-4 p-5 sm:p-6">
   446→                <div className="pointer-events-none absolute -right-10 -top-10 h-40 w-40 rounded-full bg-primary/20 blur-3xl" />
   447→                <div className="relative flex items-center gap-4">
   448→                  <div className="grid h-12 w-12 shrink-0 place-items-center rounded-xl bg-gradient-to-br from-violet-500 to-fuchsia-500 text-white shadow-lg shadow-fuchsia-500/30">
   449→                    <Repeat className="h-6 w-6" />
   450→                  </div>
   451→                  <div>
   452→                    <p className="font-semibold leading-tight">
   453→                      У тебя{' '}
   454→                      <span className="text-gradient">{dueCards.length}</span>{' '}
   455→                      {pluralizeCards(dueCards.length)} на повторение
   456→                    </p>
   457→                    <p className="mt-0.5 text-sm text-muted-foreground">
   458→                      Самое время освежить память — это занимает пару минут
   459→                    </p>
   460→                  </div>
   461→                </div>
   462→                <GradientButton onClick={startDueReview} className="shrink-0">
   463→                  <RotateCcw className="h-4 w-4" />
   464→                  Повторить
   465→                </GradientButton>
   466→              </div>
   467→            </GlassCard>
   468→          </motion.div>
   469→        )}
   470→      </AnimatePresence>
   471→
   472→      {/* Generate form */}
   473→      <GlassCard className="mb-6 overflow-hidden" hover={false}>
   474→        <div className="relative border-b border-border/60 bg-gradient-to-br from-violet-500/5 via-fuchsia-500/5 to-transparent p-5 sm:p-6">
   475→          <div className="pointer-events-none absolute inset-0 bg-grid opacity-10" />
   476→          <div className="relative">
   477→            <label
   478→              htmlFor="flash-topic"
   479→              className="mb-2 block text-sm font-medium"
   480→            >
   481→              Какую тему закрепляем?
   482→            </label>
   483→            <div className="flex flex-col gap-3 sm:flex-row">
   484→              <input
   485→                id="flash-topic"
   486→                value={topic}
   487→                onChange={(e) => setTopic(e.target.value)}
   488→                onKeyDown={(e) => {
   489→                  if (e.key === 'Enter' && !generating) handleGenerate()
   490→                }}
   491→                placeholder="Например: Бинарный поиск, irregular verbs, клеточное дыхание..."
   492→                className="flex-1 rounded-xl border border-border bg-background/60 px-4 py-3 text-sm outline-none ring-primary/20 transition placeholder:text-muted-foreground/60 focus:ring-2"
   493→              />
   494→              <GradientButton
   495→                onClick={() => handleGenerate()}
   496→                disabled={generating}
   497→                className="shrink-0 px-6"
   498→              >
   499→                {generating ? (
   500→                  <>
   501→                    <Loader2 className="h-4 w-4 animate-spin" />
   502→                    Создаю...
   503→                  </>
   504→                ) : (
   505→                  <>
   506→                    <Sparkles className="h-4 w-4" />
   507→                    Создать карты
   508→                  </>
   509→                )}
   510→              </GradientButton>
   511→            </div>
   512→          </div>
   513→        </div>
   514→
   515→        <div className="space-y-5 p-5 sm:p-6">
   516→          {/* Subject chips */}
   517→          <div>
   518→            <p className="mb-2 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   519→              Предмет
   520→            </p>
   521→            <div className="flex flex-wrap gap-2">
   522→              {SUBJECTS.map((s) => {
   523→                const active = s.id === subject
   524→                return (
   525→                  <button
   526→                    key={s.id}
   527→                    onClick={() => setSubjectLocal(s.id)}
   528→                    className={`inline-flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-xs font-medium transition-all ${
   529→                      active
   530→                        ? `border-transparent bg-gradient-to-r ${s.gradient} text-white shadow-md`
   531→                        : 'border-border/60 bg-background/60 text-muted-foreground hover:border-primary/40 hover:text-foreground'
   532→                    }`}
   533→                  >
   534→                    <span>{s.emoji}</span>
   535→                    {s.ru}
   536→                  </button>
   537→                )
   538→              })}
   539→            </div>
   540→          </div>
   541→
   542→          {/* Count selector */}
   543→          <div>
   544→            <p className="mb-2 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   545→              Сколько карточек
   546→            </p>
   547→            <div className="flex flex-wrap gap-2">
   548→              {COUNTS.map((c) => {
   549→                const active = c === count
   550→                return (
   551→                  <button
   552→                    key={c}
   553→                    onClick={() => setCount(c)}
   554→                    className={`inline-flex h-9 w-12 items-center justify-center rounded-lg border text-sm font-semibold transition-all ${
   555→                      active
   556→                        ? 'border-primary/60 bg-primary/15 text-primary ring-1 ring-primary/40'
   557→                        : 'border-border/60 bg-background/60 text-muted-foreground hover:border-primary/40 hover:text-foreground'
   558→                    }`}
   559→                  >
   560→                    {c}
   561→                  </button>
   562→                )
   563→              })}
   564→            </div>
   565→          </div>
   566→
   567→          {/* Suggested topics */}
   568→          <div>
   569→            <p className="mb-2 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   570→              Идеи тем
   571→            </p>
   572→            <div className="flex flex-wrap gap-2">
   573→              {SUGGESTED_TOPICS.map((sug) => {
   574→                const subj = getSubject(sug.subject)
   575→                return (
   576→                  <button
   577→                    key={sug.topic}
   578→                    onClick={() => {
   579→                      setSubjectLocal(sug.subject)
   580→                      handleGenerate(sug.topic)
   581→                    }}
   582→                    disabled={generating}
   583→                    className="group inline-flex items-center gap-2 rounded-full border border-border/60 bg-background/60 px-3 py-1.5 text-xs text-muted-foreground transition-colors hover:border-primary/40 hover:text-foreground disabled:opacity-50"
   584→                  >

  1034→function pluralizeCards(n: number): string {
  1035→  const mod10 = n % 10
  1036→  const mod100 = n % 100
  1037→  if (mod10 === 1 && mod100 !== 11) return 'карточка'
  1038→  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 10 || mod100 >= 20))
  1039→    return 'карточки'
  1040→  return 'карточек'
  1041→}
