    78→// ---------- main view ----------
    79→export function TutorView() {
    80→  const { activeSubject } = useNav()
    81→
    82→  const [subject, setSubject] = useState<string>(activeSubject ?? 'general')
    83→  const [sessions, setSessions] = useState<ChatSessionMeta[]>([])
    84→  const [sessionId, setSessionId] = useState<string | null>(null)
    85→  const [messages, setMessages] = useState<ChatMessage[]>([])
    86→  const [input, setInput] = useState('')
    87→  const [sending, setSending] = useState(false)
    88→  const [loadingSessions, setLoadingSessions] = useState(true)
    89→  const [loadingThread, setLoadingThread] = useState(false)
    90→  const [listOpen, setListOpen] = useState(false) // mobile sheet
    91→
    92→  const bottomRef = useRef<HTMLDivElement>(null)
    93→
    94→  // On mount: load session list, restore current session
    95→  useEffect(() => {
    96→    const stored =
    97→      typeof window !== 'undefined' ? sessionStorage.getItem(SESSION_KEY) : null
    98→    if (stored) {
    99→      void loadThread(stored)
   100→    }
   101→    void loadSessions()
   102→  }, [])
   103→
   104→  // auto-scroll to bottom when messages change / typing
   105→  useEffect(() => {
   106→    bottomRef.current?.scrollIntoView({ behavior: 'smooth' })
   107→  }, [messages, sending])
   108→
   109→  // ---------- data fns ----------
   110→  async function loadSessions() {
   111→    setLoadingSessions(true)
   112→    try {
   113→      const res = await fetch('/api/chat', { cache: 'no-store' })
   114→      const data = await res.json()
   115→      if (res.ok && Array.isArray(data.sessions)) {
   116→        setSessions(data.sessions as ChatSessionMeta[])
   117→      }
   118→    } catch {
   119→      // silent — sidebar just shows empty
   120→    } finally {
   121→      setLoadingSessions(false)
   122→    }
   123→  }
   124→
   125→  async function loadThread(sid: string) {
   126→    setLoadingThread(true)
   127→    try {
   128→      const res = await fetch(`/api/chat/${sid}`, { cache: 'no-store' })
   129→      const data = await res.json()
   130→      if (!res.ok) throw new Error(data?.error || 'Не удалось загрузить диалог')
   131→      const apiMsgs = (data.session?.messages ?? []) as ApiMessage[]
   132→      const msgs: ChatMessage[] = apiMsgs.map((m, i) => ({
   133→        id: `${sid}-${i}`,
   134→        role: m.role,
   135→        content: m.content,
   136→      }))
   137→      setMessages(msgs)
   138→      if (data.session?.subject && data.session.subject !== 'general') {
   139→        setSubject(data.session.subject)
   140→      }
   141→      setSessionId(sid)
   142→      sessionStorage.setItem(SESSION_KEY, sid)
   143→    } catch (e) {
   144→      const msg = e instanceof Error ? e.message : 'Не удалось загрузить диалог'
   145→      toast.error(msg)
   146→      setSessionId(null)
   147→      sessionStorage.removeItem(SESSION_KEY)
   148→    } finally {
   149→      setLoadingThread(false)
   150→      setListOpen(false)
   151→    }
   152→  }
   153→
   154→  async function deleteSession(sid: string, e?: React.MouseEvent) {
   155→    e?.stopPropagation()
   156→    try {
   157→      const res = await fetch(`/api/chat/${sid}`, { method: 'DELETE' })
   158→      if (!res.ok) throw new Error('Не удалось удалить')
   159→      setSessions((s) => s.filter((x) => x.id !== sid))
   160→      if (sessionId === sid) {
   161→        setSessionId(null)
   162→        setMessages([])
   163→        sessionStorage.removeItem(SESSION_KEY)
   164→      }
   165→      toast.success('Диалог удалён')
   166→    } catch {
   167→      toast.error('Не удалось удалить диалог')
   168→    }
   169→  }
   170→
   171→  function startNewChat() {
   172→    setSessionId(null)
   173→    setMessages([])
   174→    setInput('')
   175→    sessionStorage.removeItem(SESSION_KEY)
   176→    setListOpen(false)
   177→  }
   178→
   179→  // ---------- send ----------
   180→  async function send(text?: string, overrideSubject?: string) {
   181→    const content = (text ?? input).trim()
   182→    if (!content || sending) return
   183→    setInput('')
   184→
   185→    const usedSubject = overrideSubject ?? subject
   186→
   187→    const userMsg: ChatMessage = {
   188→      id: `u-${Date.now()}`,
   189→      role: 'user',
   190→      content,
   191→    }
   192→    const prevMessages = messages
   193→    const nextMessages = [...prevMessages, userMsg]
   194→    setMessages(nextMessages)
   195→    setSending(true)
   196→
   197→    try {
   198→      const history = prevMessages.map(
   199→        (m): ApiMessage => ({ role: m.role, content: m.content })
   200→      )
   201→      const res = await fetch('/api/chat', {
   202→        method: 'POST',
   203→        headers: { 'Content-Type': 'application/json' },
   204→        body: JSON.stringify({
   205→          message: content,
   206→          subject: usedSubject,
   207→          sessionId,
   208→          history,
   209→        }),
   210→      })
   211→      const data = await res.json()
   212→      if (!res.ok) throw new Error(data?.error || 'Не удалось получить ответ')
   213→
   214→      const aiMsg: ChatMessage = {
   215→        id: `a-${Date.now()}`,
   216→        role: 'assistant',
   217→        content: data.reply as string,
   218→      }
   219→      setMessages((m) => [...m, aiMsg])
   220→
   221→      const newSid: string | undefined = data.sessionId
   222→      if (newSid) {
   223→        setSessionId(newSid)
   224→        sessionStorage.setItem(SESSION_KEY, newSid)
   225→      }
   226→      if (typeof data.xp === 'number' && typeof data.level === 'number') {
   227→        useUser.setState({ xp: data.xp, level: data.level })
   228→      }
   229→      void loadSessions()
   230→    } catch (e) {
   231→      const msg = e instanceof Error ? e.message : 'Не удалось получить ответ'
   232→      toast.error(msg)
   233→    } finally {
   234→      setSending(false)
   235→    }
   236→  }
   237→
   238→  function onKeyDown(e: React.KeyboardEvent<HTMLTextAreaElement>) {
   239→    if (e.key === 'Enter' && !e.shiftKey) {
   240→      e.preventDefault()
   241→      void send()
   242→    }
   243→  }
   244→
   245→  // ---------- derived ----------
   246→  const activeSubjectObj =
   247→    subject && subject !== 'general' ? getSubject(subject) : null
   248→
   249→  return (
   250→    <PageSection className="relative py-6 sm:py-8">
   251→      {/* ambient cosmic glow */}

   253→        aria-hidden

   255→        style={{
   256→          background:
   257→            'radial-gradient(ellipse at center, oklch(0.7 0.25 315 / 0.18), transparent 70%)',
   258→        }}
   259→      />
   260→
   261→      <SectionHeader
   262→        title="AI-наставник"
   263→        subtitle="Спроси что угодно — объясню, разберу по шагам, подскажу дальше"
   264→        icon={Sparkles}
   265→        action={
   266→          <div className="hidden items-center gap-2 sm:flex">
   267→            {activeSubjectObj && (
   268→              <Pill className="border-primary/30 bg-primary/10 text-primary">
   269→                <span>{activeSubjectObj.emoji}</span>
   270→                {activeSubjectObj.ru}
   271→              </Pill>
   272→            )}
   273→          </div>
   274→        }
   275→      />
   276→
   277→      {/* mobile: open sessions sheet + new chat */}
   278→      <div className="mb-3 flex items-center gap-2 lg:hidden">
   279→        <Sheet open={listOpen} onOpenChange={setListOpen}>
   280→          <SheetTrigger asChild>
   281→            <button className="inline-flex items-center gap-2 rounded-xl border border-border/60 bg-card/60 px-3 py-2 text-sm font-medium backdrop-blur transition hover:bg-card">
   282→              <PanelLeft className="h-4 w-4" />
   283→              Диалоги
   284→              {sessions.length > 0 && (
   285→                <span className="rounded-full bg-primary/15 px-1.5 text-xs text-primary">
   286→                  {sessions.length}
   287→                </span>
   288→              )}
   289→            </button>
   290→          </SheetTrigger>
   291→          <SheetContent side="left" className="w-[85%] sm:max-w-sm">
   292→            <SheetHeader>
   293→              <SheetTitle className="flex items-center gap-2">
   294→                <MessagesSquare className="h-4 w-4 text-primary" />
   295→                Диалоги
   296→              </SheetTitle>
   297→            </SheetHeader>
   298→            <div className="flex-1 overflow-y-auto px-4 pb-6">
   299→              <SidebarContent
   300→                sessions={sessions}
   301→                loading={loadingSessions}
   302→                currentId={sessionId}
   303→                onSelect={(id) => void loadThread(id)}
   304→                onDelete={(id, ev) => void deleteSession(id, ev)}
   305→                onNew={startNewChat}
   306→              />
   307→            </div>
   308→          </SheetContent>
   309→        </Sheet>
   310→
   311→        <NewChatButton onClick={startNewChat} small />
   312→      </div>
   313→
   314→      {/* main grid */}
   315→      <div className="grid gap-4 lg:grid-cols-[320px_1fr]">
   316→        {/* left rail — desktop */}
   317→        <GlassCard
   318→          hover={false}
   319→          className="hidden h-[72vh] min-h-[520px] flex-col overflow-hidden p-3 lg:flex lg:h-[calc(100vh-240px)]"
   320→        >
   321→          <SidebarContent
   322→            sessions={sessions}
   323→            loading={loadingSessions}
   324→            currentId={sessionId}
   325→            onSelect={(id) => void loadThread(id)}
   326→            onDelete={(id, ev) => void deleteSession(id, ev)}
   327→            onNew={startNewChat}
   328→          />
   329→        </GlassCard>
   330→
   331→        {/* right — chat thread + composer */}
   332→        <GlassCard
   333→          hover={false}
   334→          gradient
   335→          className="flex h-[72vh] min-h-[520px] flex-col overflow-hidden lg:h-[calc(100vh-240px)]"
   336→        >
   337→          {/* chat header */}
   338→          <div className="flex items-center justify-between gap-3 border-b border-border/60 bg-gradient-to-r from-violet-500/5 via-fuchsia-500/5 to-pink-500/5 px-4 py-3">
   339→            <div className="flex min-w-0 items-center gap-3">
   340→              <div className="relative grid h-9 w-9 shrink-0 place-items-center rounded-xl bg-gradient-to-br from-violet-500 via-fuchsia-500 to-pink-500 text-white shadow-lg shadow-fuchsia-500/30">
   341→                <Sparkles className="h-4 w-4" />
   342→                <span className="absolute -bottom-0.5 -right-0.5 h-2.5 w-2.5 rounded-full bg-emerald-400 ring-2 ring-card" />
   343→              </div>
   344→              <div className="min-w-0">
   345→                <p className="truncate text-sm font-semibold leading-tight">
   346→                  {sessionId ? 'Активный диалог' : 'Новый диалог'}
   347→                </p>
   348→                <p className="truncate text-xs text-muted-foreground">
   349→                  {sending ? 'Наставник печатает…' : 'Готов помочь'}
   350→                </p>
   351→              </div>
   352→            </div>
   353→
   354→            <SubjectPicker value={subject} onChange={setSubject} />
   355→          </div>
   356→
   357→          {/* messages area */}
   358→          <div className="relative flex-1 overflow-hidden">
   359→            <ScrollArea className="h-full">
   360→              <div className="space-y-4 px-4 py-4 sm:px-6">
   361→                {loadingThread ? (
   362→                  <LoadingState label="Загружаю диалог…" />
   363→                ) : messages.length === 0 ? (
   364→                  <EmptyChat
   365→                    onPick={(p, s) => {
   366→                      if (s) setSubject(s)
   367→                      void send(p, s)
   368→                    }}
   369→                  />
   370→                ) : (
   371→                  <AnimatePresence initial={false}>
   372→                    {messages.map((m) => (
   373→                      <MessageBubble key={m.id} message={m} />
   374→                    ))}
   375→                  </AnimatePresence>
   376→                )}
   377→
   378→                {sending && <TypingBubble />}
   379→
   380→                <div ref={bottomRef} className="h-1" />
   381→              </div>
   382→            </ScrollArea>
   383→          </div>
   384→
   385→          {/* composer */}
   386→          <Composer
   387→            value={input}
   388→            onChange={setInput}
   389→            onKeyDown={onKeyDown}
   390→            onSend={() => void send()}
   391→            disabled={sending}
   392→          />
   393→        </GlassCard>
   394→      </div>
   395→    </PageSection>
   396→  )
   397→}

   399→// ---------- sidebar content ----------
   400→function SidebarContent({
   401→  sessions,
   402→  loading,
   403→  currentId,
   404→  onSelect,
   405→  onDelete,
   406→  onNew,
   407→}: {
   408→  sessions: ChatSessionMeta[]
   409→  loading: boolean
   410→  currentId: string | null
   411→  onSelect: (id: string) => void
   412→  onDelete: (id: string, e?: React.MouseEvent) => void
   413→  onNew: () => void
   414→}) {
   415→  return (
   416→    <div className="flex h-full flex-col gap-3">
   417→      <NewChatButton onClick={onNew} />
   418→
   419→      <div className="flex items-center gap-2 px-1 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   420→        <MessagesSquare className="h-3.5 w-3.5" />
   421→        Недавние
   422→      </div>
   423→
   424→      <div className="-mr-1 flex-1 overflow-y-auto pr-1">
   425→        {loading ? (
   426→          <div className="space-y-2">
   427→            {Array.from({ length: 4 }).map((_, i) => (
   428→              <div key={i} className="h-14 animate-pulse rounded-xl bg-muted/40" />
   429→            ))}
   430→          </div>
   431→        ) : sessions.length === 0 ? (
   432→          <div className="rounded-xl border border-dashed border-border/60 p-4 text-center text-xs text-muted-foreground">
   433→            Пока нет диалогов

   435→            Задай первый вопрос 👇

   438→          <ul className="space-y-1.5">
   439→            {sessions.map((s) => {
   440→              const subj = getSubject(s.subject)
   441→              const active = s.id === currentId
   442→              return (
   443→                <li key={s.id}>
   444→                  <button
   445→                    onClick={() => onSelect(s.id)}
   446→                    className={cn(
   447→                      'group relative flex w-full items-start gap-2.5 rounded-xl border p-2.5 text-left transition-all',
   448→                      active
   449→                        ? 'border-primary/40 bg-primary/10 glow-soft'
   450→                        : 'border-border/50 bg-background/40 hover:border-border hover:bg-card'
   451→                    )}
   452→                  >
   453→                    <div
   454→                      className={cn(
   455→                        'mt-0.5 grid h-7 w-7 shrink-0 place-items-center rounded-lg bg-gradient-to-br text-xs',
   456→                        subj ? subj.gradient : 'from-violet-500 to-fuchsia-500'
   457→                      )}
   458→                    >
   459→                      <span className="text-white">{subj?.emoji ?? '💬'}</span>
   460→                    </div>
   461→                    <div className="min-w-0 flex-1">
   462→                      <p className="truncate text-sm font-medium leading-tight">
   463→                        {s.title || 'Без названия'}
   464→                      </p>
   465→                      <p className="mt-0.5 truncate text-xs text-muted-foreground">
   466→                        {s._count?.messages ?? 0} сообщ. · {formatDate(s.updatedAt)}
   467→                      </p>
   468→                    </div>
   469→                    <button
   470→                      onClick={(e) => onDelete(s.id, e)}
   471→                      className="absolute right-1.5 top-1.5 rounded-md p-1 text-muted-foreground opacity-0 transition-all hover:bg-destructive/15 hover:text-destructive group-hover:opacity-100"
   472→                      aria-label="Удалить диалог"
   473→                    >
   474→                      <Trash2 className="h-3.5 w-3.5" />
   475→                    </button>
   476→                  </button>
   477→                </li>
   478→              )
   479→            })}
   480→          </ul>
   481→        )}
   482→      </div>
   483→    </div>
   484→  )
   485→}

   551→  const isUser = message.role === 'user'
   552→  return (
   553→    <motion.div
   554→      layout
   555→      initial={{ opacity: 0, y: 12 }}

   558→      className={cn('flex gap-3', isUser ? 'flex-row-reverse' : 'flex-row')}
   559→    >
   560→      {/* avatar */}
   561→      {isUser ? (
   562→        <div className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border/60 bg-card/60 text-muted-foreground">

   565→      ) : (
   566→        <div className="grid h-8 w-8 shrink-0 place-items-center rounded-full bg-gradient-to-br from-violet-500 via-fuchsia-500 to-pink-500 text-white shadow-md shadow-fuchsia-500/30">

   570→
   571→      {/* bubble */}

   719→// ---------- utils ----------
   720→function formatDate(iso: string): string {
   721→  try {
   722→    const d = new Date(iso)
   723→    const now = new Date()
   724→    const diffMs = now.getTime() - d.getTime()
   725→    const diffMin = Math.floor(diffMs / 60000)
   726→    if (diffMin < 1) return 'только что'
   727→    if (diffMin < 60) return `${diffMin} мин назад`
   728→    const diffH = Math.floor(diffMin / 60)
   729→    if (diffH < 24) return `${diffH} ч назад`
   730→    const diffD = Math.floor(diffH / 24)
   731→    if (diffD < 7) return `${diffD} д назад`
   732→    return d.toLocaleDateString('ru-RU', { day: 'numeric', month: 'short' })
   733→  } catch {
   734→    return ''
   735→  }
   736→}
