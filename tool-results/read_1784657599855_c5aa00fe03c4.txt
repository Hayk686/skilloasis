   191→  count: number

   194→
   195→function build7DaySeries(byDay: Record<string, number>): DaySeriesItem[] {
   196→  const days: DaySeriesItem[] = []
   197→  const today = new Date()
   198→  for (let i = 6; i >= 0; i--) {
   199→    const d = new Date(today)
   200→    d.setDate(today.getDate() - i)
   201→    const key = d.toISOString().slice(0, 10)
   202→    const weekday = d.toLocaleDateString('ru-RU', { weekday: 'short' })
   203→    const isToday = i === 0
   204→    days.push({
   205→      key,
   206→      label: weekday.charAt(0).toUpperCase() + weekday.slice(1),
   207→      count: byDay[key] || 0,
   208→      isToday,
   209→    })
   210→  }
   211→  return days
   212→}

   277→          cx="64"

   281→          stroke="currentColor"

   312→/* ============================================================ *
   313→ * Confetti burst
   314→ * ============================================================ */
   315→
   316→function ConfettiBurst({ burst }: { burst: number }) {
   317→  const particles = useMemo(() => {
   318→    if (burst === 0) return []
   319→    return Array.from({ length: 28 }, (_, i) => {
   320→      const angle = (i / 28) * Math.PI * 2 + Math.random() * 0.4
   321→      const distance = 90 + Math.random() * 110
   322→      return {
   323→        id: `${burst}-${i}`,
   324→        x: Math.cos(angle) * distance,
   325→        y: Math.sin(angle) * distance,
   326→        delay: Math.random() * 0.12,
   327→        emoji: ['🎉', '✨', '⭐', '🔥', '💫', '🌟'][i % 6],
   328→        rotate: (Math.random() - 0.5) * 540,
   329→      }
   330→    })
   331→  }, [burst])
   332→
   333→  return (
   334→    <div className="pointer-events-none absolute inset-0 flex items-center justify-center overflow-visible">
   335→      <AnimatePresence>
   336→        {particles.map((p) => (
   337→          <motion.div
   338→            key={p.id}
   339→            className="absolute text-lg"
   340→            initial={{ x: 0, y: 0, opacity: 1, scale: 0, rotate: 0 }}
   341→            animate={{
   342→              x: p.x,
   343→              y: p.y,
   344→              opacity: 0,
   345→              scale: 1.1,
   346→              rotate: p.rotate,
   347→            }}
   348→            transition={{ duration: 1.5, delay: p.delay, ease: 'easeOut' }}
   349→          >
   350→            {p.emoji}
   351→          </motion.div>
   352→        ))}
   353→      </AnimatePresence>
   354→    </div>
   355→  )
   356→}

   446→
   447→function ChartTooltip({ active, payload, label }: TooltipProps<number, string>) {
   448→  if (!active || !payload || payload.length === 0) return null
   449→  const v = Number(payload[0]?.value ?? 0)
   450→  return (
   451→    <div className="rounded-xl border border-border/70 bg-card/95 px-3 py-2 text-xs shadow-xl backdrop-blur-md">
   452→      <div className="font-semibold text-foreground">{label}</div>
   453→      <div className="mt-0.5 text-muted-foreground">
   454→        {v === 0 ? 'нет активности' : `${v} ${pluralActivity(v)}`}
   455→      </div>
   456→    </div>
   457→  )
   458→}
   459→
   460→function pluralActivity(n: number) {
   461→  const mod10 = n % 10
   462→  const mod100 = n % 100
   463→  if (mod10 === 1 && mod100 !== 11) return 'действие'
   464→  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 10 || mod100 >= 20))
   465→    return 'действия'
   466→  return 'действий'
   467→}

   492→  const [hintOpen, setHintOpen] = useState(false)
   493→  const [confettiBurst, setConfettiBurst] = useState(0)

   500→      setLoading(true)

   502→        const [uRes, pRes, dRes] = await Promise.all([
   503→          fetch('/api/user', { cache: 'no-store' }),

   609→      toast.error('Не удалось засчитать вызов')
   610→    } finally {
   611→      setSubmittingDaily(false)
   612→    }
   613→  }, [challenge, dailyDone, submittingDaily])

   627→    () => daySeries.reduce((s, d) => s + d.count, 0),

  1443→function pluralDays(n: number) {
  1444→  const mod10 = n % 10
  1445→  const mod100 = n % 100
  1446→  if (mod10 === 1 && mod100 !== 11) return 'день'
