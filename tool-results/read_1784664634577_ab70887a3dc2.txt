    81→    81→function difficultyMeta(d: string) {
    82→    82→  const diff = (d || '').toLowerCase()
    83→    83→  if (
    84→    84→    diff.startsWith('easy') ||
    85→    85→    diff.startsWith('лёг') ||
    86→    86→    diff.startsWith('лег') ||
    87→    87→    diff.startsWith('прост')
    88→    88→  ) {
    89→    89→    return {
    90→    90→      label: 'Легко',
    91→    91→      cls: 'border-emerald-500/40 bg-emerald-500/10 text-emerald-300',
    92→    92→      dot: 'bg-emerald-400',
    93→    93→    }
    94→    94→  }
    95→    95→  if (
    96→    96→    diff.startsWith('hard') ||
    97→    97→    diff.startsWith('слож') ||
    98→    98→    diff.startsWith('тяж') ||
    99→    99→    diff.startsWith('прод')
   100→   100→  ) {
   101→   101→    return {
   102→   102→      label: 'Сложно',
   103→   103→      cls: 'border-rose-500/40 bg-rose-500/10 text-rose-300',
   104→   104→      dot: 'bg-rose-400',
   105→   105→    }
   106→   106→  }
   107→   107→  return {
   108→   108→    label: 'Средне',
   109→   109→    cls: 'border-amber-500/40 bg-amber-500/10 text-amber-300',
   110→   110→    dot: 'bg-amber-400',
   111→   111→  }
   112→   112→}

   114→   114→function formatTime(sec: number) {
   115→   115→  const m = Math.floor(sec / 60)
   116→   116→  const s = sec % 60
   117→   117→  return `${m}:${s.toString().padStart(2, '0')}`
   118→   118→}

   122→   122→export function QuizView() {
   123→   123→  const { activeSubject } = useNav()
   124→   124→  const [phase, setPhase] = useState<Phase>('setup')
   125→   125→  const [topic, setTopic] = useState('')
   126→   126→  const [subject, setSubject] = useState<string>(activeSubject || 'programming')
   127→   127→  const [count, setCount] = useState<number>(5)
   128→   128→  const [level, setLevel] = useState<string>('Любой')
   129→   129→  const [questions, setQuestions] = useState<QuizQuestion[]>([])
   130→   130→  const [answers, setAnswers] = useState<(number | null | undefined)[]>([])
   131→   131→  const [currentIdx, setCurrentIdx] = useState(0)
   132→   132→  const [revealed, setRevealed] = useState(false)
   133→   133→  const [elapsed, setElapsed] = useState(0)
   134→   134→  const [quizResult, setQuizResult] = useState<QuizResult | null>(null)
   135→   135→  const [submitting, setSubmitting] = useState(false)
   136→   136→
   137→   137→  // Prefill subject from nav if available
   138→   138→  useEffect(() => {
   139→   139→    if (activeSubject) setSubject(activeSubject)
   140→   140→  }, [activeSubject])
   141→   141→
   142→   142→  // Per-question elapsed timer (stops once revealed)
   143→   143→  useEffect(() => {
   144→   144→    setElapsed(0)
   145→   145→  }, [currentIdx])
   146→   146→  useEffect(() => {
   147→   147→    if (phase !== 'quiz' || revealed) return
   148→   148→    const id = setInterval(() => setElapsed((e) => e + 1), 1000)
   149→   149→    return () => clearInterval(id)
   150→   150→  }, [phase, currentIdx, revealed])
   151→   151→
   152→   152→  const suggestedTopics = useMemo(() => {
   153→   153→    const subj = getSubject(subject)
   154→   154→    return subj?.topics?.slice(0, 6) ?? FALLBACK_TOPICS
   155→   155→  }, [subject])
   156→   156→
   157→   157→  const currentScore = useMemo(() => {
   158→   158→    return questions.reduce((acc, q, i) => {
   159→   159→      const isRevealed = i < currentIdx || (i === currentIdx && revealed)
   160→   160→      if (isRevealed && answers[i] === q.correctIndex) return acc + 1
   161→   161→      return acc
   162→   162→    }, 0)
   163→   163→  }, [questions, currentIdx, revealed, answers])
   164→   164→
   165→   165→  const addressedCount = useMemo(
   166→   166→    () => answers.filter((a) => a !== undefined).length,
   167→   167→    [answers]
   168→   168→  )

   170→   170→  async function startQuiz(opts?: {
   171→   171→    topic?: string
   172→   172→    subject?: string
   173→   173→    count?: number
   174→   174→    level?: string
   175→   175→  }) {
   176→   176→    const t = (opts?.topic ?? topic).trim()
   177→   177→    const s = opts?.subject ?? subject
   178→   178→    const c = opts?.count ?? count
   179→   179→    const l = opts?.level ?? level
   180→   180→    if (!t) {
   181→   181→      toast.error('Укажите тему квиза')
   182→   182→      return
   183→   183→    }
   184→   184→    setTopic(t)
   185→   185→    setSubject(s)
   186→   186→    setCount(c)
   187→   187→    setLevel(l)
   188→   188→    setPhase('loading')
   189→   189→    setQuizResult(null)
   190→   190→    try {
   191→   191→      const res = await fetch('/api/quiz', {
   192→   192→        method: 'POST',
   193→   193→        headers: { 'Content-Type': 'application/json' },
   194→   194→        body: JSON.stringify({
   195→   195→          topic: t,
   196→   196→          subject: s,
   197→   197→          count: c,
   198→   198→          level: l === 'Любой' ? undefined : l,
   199→   199→        }),
   200→   200→      })
   201→   201→      const data = await res.json()
   202→   202→      if (!res.ok) throw new Error(data.error)
   203→   203→      if (!data.questions?.length) throw new Error('Нет вопросов')
   204→   204→      setQuestions(data.questions)
   205→   205→      setAnswers(Array(data.questions.length).fill(undefined))
   206→   206→      setCurrentIdx(0)
   207→   207→      setRevealed(false)
   208→   208→      setElapsed(0)
   209→   209→      setPhase('quiz')
   210→   210→      toast.success('Квиз готов! Удачи 🎯')
   211→   211→    } catch {
   212→   212→      toast.error('Не удалось создать квиз. Попробуйте ещё раз.')
   213→   213→      setPhase('setup')
   214→   214→    }
   215→   215→  }
   216→   216→
   217→   217→  function selectOption(idx: number) {
   218→   218→    if (revealed) return
   219→   219→    const next = [...answers]
   220→   220→    next[currentIdx] = idx
   221→   221→    setAnswers(next)
   222→   222→  }
   223→   223→
   224→   224→  function handleCheck() {
   225→   225→    if (answers[currentIdx] === undefined || answers[currentIdx] === null) {
   226→   226→      toast.error('Выбери вариант ответа')
   227→   227→      return
   228→   228→    }
   229→   229→    setRevealed(true)
   230→   230→  }
   231→   231→
   232→   232→  function handleSkip() {
   233→   233→    if (revealed) return
   234→   234→    const next = [...answers]
   235→   235→    next[currentIdx] = null
   236→   236→    setAnswers(next)
   237→   237→    setRevealed(true)
   238→   238→  }
   239→   239→
   240→   240→  async function handleNext() {
   241→   241→    if (currentIdx + 1 >= questions.length) {
   242→   242→      await finishQuiz()
   243→   243→    } else {
   244→   244→      setCurrentIdx(currentIdx + 1)
   245→   245→      setRevealed(false)
   246→   246→    }
   247→   247→  }
   248→   248→
   249→   249→  async function finishQuiz() {
   250→   250→    const correct = questions.reduce(
   251→   251→      (acc, q, i) => acc + (answers[i] === q.correctIndex ? 1 : 0),
   252→   252→      0
   253→   253→    )
   254→   254→    setPhase('results')
   255→   255→    setSubmitting(true)
   256→   256→    try {
   257→   257→      const res = await fetch('/api/quiz', {
   258→   258→        method: 'PUT',
   259→   259→        headers: { 'Content-Type': 'application/json' },
   260→   260→        body: JSON.stringify({
   261→   261→          subject,
   262→   262→          topic,
   263→   263→          total: questions.length,
   264→   264→          correct,
   265→   265→        }),
   266→   266→      })
   267→   267→      const data = await res.json()
   268→   268→      if (!res.ok) throw new Error(data.error)
   269→   269→      setQuizResult(data)
   270→   270→      useUser.setState({ xp: data.xp, level: data.level })
   271→   271→      if (data.xpGain > 0) {
   272→   272→        toast.success(`+${data.xpGain} XP`, { description: 'Прогресс сохранён' })
   273→   273→      }
   274→   274→    } catch {
   275→   275→      toast.error('Не удалось сохранить прогресс')
   276→   276→    } finally {
   277→   277→      setSubmitting(false)
   278→   278→    }
   279→   279→  }
   280→   280→
   281→   281→  function resetToSetup(clearTopic = true) {
   282→   282→    setPhase('setup')
   283→   283→    setQuestions([])
   284→   284→    setAnswers([])
   285→   285→    setCurrentIdx(0)
   286→   286→    setRevealed(false)
   287→   287→    setQuizResult(null)
   288→   288→    if (clearTopic) setTopic('')
   289→   289→  }
   290→   290→
   291→   291→  return (
   292→   292→    <PageSection className="py-8">
   293→   293→      <SectionHeader
   294→   294→        title="Квиз-арена"
   295→   295→        subtitle="Адаптивный квиз по любой теме — проверь себя и заработай XP"
   296→   296→        icon={Trophy}
   297→   297→      />
   298→   298→
   299→   299→      <AnimatePresence mode="wait">
   300→   300→        {phase === 'setup' && (
   301→   301→          <motion.div
   302→   302→            key="setup"
   303→   303→            initial={{ opacity: 0, y: 12 }}
   304→   304→            animate={{ opacity: 1, y: 0 }}
   305→   305→            exit={{ opacity: 0, y: -12 }}
   306→   306→            transition={{ duration: 0.3 }}
   307→   307→          >
   308→   308→            <SetupPhase
   309→   309→              topic={topic}
   310→   310→              setTopic={setTopic}
   311→   311→              subject={subject}
   312→   312→              setSubject={setSubject}
   313→   313→              count={count}
   314→   314→              setCount={setCount}
   315→   315→              level={level}
   316→   316→              setLevel={setLevel}
   317→   317→              suggestedTopics={suggestedTopics}
   318→   318→              onStart={() => startQuiz()}
   319→   319→            />
   320→   320→          </motion.div>
   321→   321→        )}
   322→   322→
   323→   323→        {phase === 'loading' && (
   324→   324→          <motion.div
   325→   325→            key="loading"
   326→   326→            initial={{ opacity: 0 }}
   327→   327→            animate={{ opacity: 1 }}
   328→   328→            exit={{ opacity: 0 }}
   329→   329→          >
   330→   330→            <QuizSkeleton />
   331→   331→          </motion.div>
   332→   332→        )}
   333→   333→
   334→   334→        {phase === 'quiz' && questions.length > 0 && (
   335→   335→          <motion.div
   336→   336→            key="quiz"
   337→   337→            initial={{ opacity: 0 }}
   338→   338→            animate={{ opacity: 1 }}
   339→   339→            exit={{ opacity: 0 }}
   340→   340→          >
   341→   341→            <QuizPhase
   342→   342→              question={questions[currentIdx]}
   343→   343→              questionNumber={currentIdx + 1}
   344→   344→              total={questions.length}
   345→   345→              selected={answers[currentIdx]}
   346→   346→              revealed={revealed}
   347→   347→              score={currentScore}
   348→   348→              elapsed={elapsed}
   349→   349→              progressValue={(addressedCount / questions.length) * 100}
   350→   350→              onSelect={selectOption}
   351→   351→              onCheck={handleCheck}
   352→   352→              onSkip={handleSkip}
   353→   353→              onNext={handleNext}
   354→   354→              onExit={() => resetToSetup(true)}
   355→   355→              isLast={currentIdx + 1 >= questions.length}
   356→   356→            />
   357→   357→          </motion.div>
   358→   358→        )}
   359→   359→
   360→   360→        {phase === 'results' && (
   361→   361→          <motion.div
   362→   362→            key="results"
   363→   363→            initial={{ opacity: 0 }}
   364→   364→            animate={{ opacity: 1 }}
   365→   365→            exit={{ opacity: 0 }}
   366→   366→          >
   367→   367→            <ResultsPhase
   368→   368→              questions={questions}
   369→   369→              answers={answers}
   370→   370→              topic={topic}
   371→   371→              subject={subject}
   372→   372→              result={quizResult}
   373→   373→              submitting={submitting}
   374→   374→              onNewQuiz={() => resetToSetup(true)}
   375→   375→              onRetryTopic={() => startQuiz({ topic })}
   376→   376→            />
   377→   377→          </motion.div>
   378→   378→        )}
   379→   379→      </AnimatePresence>
   380→   380→    </PageSection>
   381→   381→  )
   382→   382→}
