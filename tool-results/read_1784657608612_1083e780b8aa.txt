    77→    77→
    78→    78→export function LessonsView() {
    79→    79→  const { activeSubject } = useNav()
    80→    80→  const [topic, setTopic] = useState('')
    81→    81→  const [subject, setSubject] = useState<string | null>(activeSubject)
    82→    82→  const [level, setLevel] = useState<string>('Новичок')
    83→    83→  const [lesson, setLesson] = useState<Lesson | null>(null)
    84→    84→  const [loading, setLoading] = useState(false)
    85→    85→  const [progress, setProgress] = useState(0)
    86→    86→  const lessonRef = useRef<HTMLDivElement>(null)
    87→    87→  const inputRef = useRef<HTMLTextAreaElement>(null)
    88→    88→
    89→    89→  // On mount: pick up handoff topic from sessionStorage (set by Paths view)
    90→    90→  useEffect(() => {
    91→    91→    if (typeof window === 'undefined') return
    92→    92→    const stored = window.sessionStorage.getItem('lumina:lesson-topic')
    93→    93→    if (stored) {
    94→    94→      window.sessionStorage.removeItem('lumina:lesson-topic')
    95→    95→      setTopic(stored)
    96→    96→      // Fire generation on next tick so state settles
    97→    97→      setTimeout(() => {
    98→    98→        void generate(stored)
    99→    99→      }, 60)
   100→   100→    }
   101→   101→    // Mount-only effect: picks up handoff topic from sessionStorage.
   102→   102→  }, [])
   103→   103→
   104→   104→  // Sync subject from nav when it changes
   105→   105→  useEffect(() => {
   106→   106→    if (activeSubject) setSubject(activeSubject)
   107→   107→  }, [activeSubject])
   108→   108→
   109→   109→  // Reading progress: track scroll position relative to lesson container
   110→   110→  useEffect(() => {
   111→   111→    if (!lesson) {
   112→   112→      setProgress(0)
   113→   113→      return
   114→   114→    }
   115→   115→    function onScroll() {
   116→   116→      const el = lessonRef.current
   117→   117→      if (!el) return
   118→   118→      const rect = el.getBoundingClientRect()
   119→   119→      const total = Math.max(el.offsetHeight - window.innerHeight, 1)
   120→   120→      const scrolled = Math.max(-rect.top, 0)
   121→   121→      const pct = Math.min(100, Math.max(0, (scrolled / total) * 100))
   122→   122→      setProgress(pct)
   123→   123→    }
   124→   124→    window.addEventListener('scroll', onScroll, { passive: true })
   125→   125→    window.addEventListener('resize', onScroll)
   126→   126→    onScroll()
   127→   127→    return () => {
   128→   128→      window.removeEventListener('scroll', onScroll)
   129→   129→      window.removeEventListener('resize', onScroll)
   130→   130→    }
   131→   131→  }, [lesson])
   132→   132→
   133→   133→  const generate = useCallback(
   134→   134→    async (topicArg?: string) => {
   135→   135→      const finalTopic = (topicArg ?? topic).trim()
   136→   136→      if (!finalTopic) {
   137→   137→        toast.error('Введи тему урока')
   138→   138→        inputRef.current?.focus()
   139→   139→        return
   140→   140→      }
   141→   141→      setTopic(finalTopic)
   142→   142→      setLoading(true)
   143→   143→      try {
   144→   144→        const res = await fetch('/api/lesson', {
   145→   145→          method: 'POST',
   146→   146→          headers: { 'Content-Type': 'application/json' },
   147→   147→          body: JSON.stringify({
   148→   148→            topic: finalTopic,
   149→   149→            subject: subject ?? 'general',
   150→   150→            level,
   151→   151→          }),
   152→   152→        })
   153→   153→        const data = await res.json()
   154→   154→        if (!res.ok) {
   155→   155→          throw new Error(data?.error || 'Не удалось создать урок')
   156→   156→        }
   157→   157→        setLesson(data.lesson as Lesson)
   158→   158→        if (typeof data.xp === 'number') {
   159→   159→          useUser.setState({ xp: data.xp, level: data.level })
   160→   160→        }
   161→   161→        toast.success('Урок готов! +25 XP')
   162→   162→        // Scrolling is handled by the dedicated effect below (fires when lesson
   163→   163→        // commits to the DOM, so the ref is reliably attached).
   164→   164→      } catch (e) {
   165→   165→        const msg = e instanceof Error ? e.message : 'Не удалось создать урок'
   166→   166→        toast.error(msg)
   167→   167→      } finally {
   168→   168→        setLoading(false)
   169→   169→      }
   170→   170→    },
   171→   171→    [topic, subject, level]
   172→   172→  )
   173→   173→
   174→   174→  const resetToNew = useCallback(() => {
   175→   175→    setLesson(null)
   176→   176→    setTopic('')
   177→   177→    setProgress(0)
   178→   178→    setTimeout(() => inputRef.current?.focus(), 80)
   179→   179→  }, [])
   180→   180→
   181→   181→  // Smooth-scroll to the lesson reader whenever a new lesson finishes loading.
   182→   182→  // Using an effect (rather than a setTimeout inside generate) guarantees the
   183→   183→  // ref is attached to the freshly-mounted DOM node.
   184→   184→  useEffect(() => {
   185→   185→    if (!lesson || loading) return
   186→   186→    const t = window.setTimeout(() => {
   187→   187→      lessonRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   188→   188→    }, 80)
   189→   189→    return () => window.clearTimeout(t)
   190→   190→  }, [lesson, loading])
   191→   191→
   192→   192→  const subjectObj = subject ? getSubject(subject) : undefined
   193→   193→
   194→   194→  return (
   195→   195→    <PageSection className="py-8">
   196→   196→      {/* Reading progress bar — fixed below the topbar */}
   197→   197→      <AnimatePresence>
   198→   198→        {lesson && (
   199→   199→          <motion.div
   200→   200→            key="progress-bar"
   201→   201→            initial={{ opacity: 0 }}
   202→   202→            animate={{ opacity: 1 }}
   203→   203→            exit={{ opacity: 0 }}
   204→   204→            className="fixed left-0 right-0 top-16 z-30 h-1"
   205→   205→          >
   206→   206→            <div
   207→   207→              className="h-full bg-gradient-to-r from-violet-500 via-fuchsia-500 to-pink-500 transition-[width] duration-150 ease-out"
   208→   208→              style={{ width: `${progress}%` }}
   209→   209→            />
   210→   210→          </motion.div>
   211→   211→        )}
   212→   212→      </AnimatePresence>
   213→   213→
   214→   214→      <SectionHeader
   215→   215→        title="Интерактивные уроки"
   216→   216→        subtitle="Введи тему — AI создаст красивый урок за секунды"
   217→   217→        icon={BookOpen}
   218→   218→      />
   219→   219→
   220→   220→      {/* Input panel: full or compact */}
   221→   221→      {/* Saved lessons */}
   222→   222→      <SavedLessonsBar onOpen={(t, s) => { setTopic(t); setSubject(s); generate(t) }} />
   223→   223→
   224→   224→      <div className="mb-6">
   225→   225→        <AnimatePresence mode="wait">
   226→   226→          {lesson ? (
   227→   227→            <motion.div
   228→   228→              key="compact-bar"
   229→   229→              initial={{ opacity: 0, y: -8 }}
   230→   230→              animate={{ opacity: 1, y: 0 }}
   231→   231→              exit={{ opacity: 0, y: -8 }}
   232→   232→              transition={{ duration: 0.2 }}
   233→   233→            >
   234→   234→              <CompactBar
   235→   235→                lesson={lesson}
   236→   236→                onNew={resetToNew}
   237→   237→                subjectLabel={subjectObj?.ru}
   238→   238→                onScrollToLesson={() =>
   239→   239→                  lessonRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   240→   240→                }
   241→   241→              />
   242→   242→            </motion.div>
   243→   243→          ) : (
   244→   244→            <motion.div
   245→   245→              key="full-input"
   246→   246→              initial={{ opacity: 0, y: -8 }}
   247→   247→              animate={{ opacity: 1, y: 0 }}
   248→   248→              exit={{ opacity: 0, y: -8 }}
   249→   249→              transition={{ duration: 0.2 }}
   250→   250→            >
   251→   251→              <FullInputPanel
   252→   252→                topic={topic}
   253→   253→                setTopic={setTopic}
   254→   254→                subject={subject}
   255→   255→                setSubject={setSubject}
   256→   256→                level={level}
   257→   257→                setLevel={setLevel}
   258→   258→                loading={loading}
   259→   259→                onGenerate={generate}
   260→   260→                inputRef={inputRef}
   261→   261→              />
   262→   262→            </motion.div>
   263→   263→          )}
   264→   264→        </AnimatePresence>
   265→   265→      </div>
   266→   266→
   267→   267→      {/* Skeleton while generating */}
   268→   268→      <AnimatePresence>
   269→   269→        {loading && (
   270→   270→          <motion.div
   271→   271→            key="skeleton"
   272→   272→            initial={{ opacity: 0, y: 8 }}
   273→   273→            animate={{ opacity: 1, y: 0 }}
   274→   274→            exit={{ opacity: 0 }}
   275→   275→            transition={{ duration: 0.25 }}
   276→   276→          >
   277→   277→            <LessonSkeleton />
   278→   278→          </motion.div>
   279→   279→        )}
   280→   280→      </AnimatePresence>
   281→   281→
   282→   282→      {/* Lesson reader — keyed by title so a new lesson re-mounts & replays
   283→   283→          the reveal animation. No AnimatePresence here so the new node mounts
   284→   284→          immediately (keeping the scroll-to-lesson ref reliable). */}
   285→   285→      {lesson && !loading && (
   286→   286→        <motion.div
   287→   287→          key={`reader-${lesson.title}`}
   288→   288→          ref={lessonRef}
   289→   289→          initial={{ opacity: 0, y: 12 }}
   290→   290→          animate={{ opacity: 1, y: 0 }}
   291→   291→          transition={{ duration: 0.45, ease: 'easeOut' }}
   292→   292→        >
   293→   293→          <LessonReader
   294→   294→            lesson={lesson}
   295→   295→            subjectLabel={subjectObj?.ru}
   296→   296→            onNext={(t) => generate(t)}
   297→   297→          />
   298→   298→        </motion.div>
   299→   299→      )}
   300→   300→
   301→   301→      {/* Empty state with suggested topics */}
   302→   302→      <AnimatePresence>
   303→   303→        {!lesson && !loading && (
   304→   304→          <motion.div
   305→   305→            key="empty"
   306→   306→            initial={{ opacity: 0 }}
   307→   307→            animate={{ opacity: 1 }}
   308→   308→            exit={{ opacity: 0 }}
   309→   309→          >
   310→   310→            <EmptyState
   311→   311→              icon={Sparkles}
   312→   312→              title="Идеи для изучения"
   313→   313→              description="Введи свою тему выше или начни с одной из этих"
   314→   314→              action={
   315→   315→                <div className="mt-2 flex flex-wrap justify-center gap-2">
   316→   316→                  {SUGGESTED_TOPICS.map((t) => (
   317→   317→                    <button
   318→   318→                      key={t}
   319→   319→                      onClick={() => generate(t)}
   320→   320→                      className="group inline-flex items-center gap-1.5 rounded-full border border-border/60 bg-background/60 px-3.5 py-1.5 text-xs font-medium text-muted-foreground transition-all hover:border-primary/40 hover:text-foreground"
   321→   321→                    >
   322→   322→                      {t}
   323→   323→                      <ChevronRight className="h-3 w-3 -translate-x-1 opacity-0 transition-all group-hover:translate-x-0 group-hover:opacity-100" />
   324→   324→                    </button>
   325→   325→                  ))}
   326→   326→                </div>
   327→   327→              }
   328→   328→            />
   329→   329→          </motion.div>
   330→   330→        )}
   331→   331→      </AnimatePresence>
   332→   332→    </PageSection>
   333→   333→  )
   334→   334→}

   337→   337→
   338→   338→function FullInputPanel({
   339→   339→  topic,
   340→   340→  setTopic,
   341→   341→  subject,
   342→   342→  setSubject,
   343→   343→  level,
   344→   344→  setLevel,
   345→   345→  loading,
   346→   346→  onGenerate,
   347→   347→  inputRef,
   348→   348→}: {
   349→   349→  topic: string
   350→   350→  setTopic: (v: string) => void
   351→   351→  subject: string | null
   352→   352→  setSubject: (v: string | null) => void
   353→   353→  level: string
   354→   354→  setLevel: (v: string) => void
   355→   355→  loading: boolean
   356→   356→  onGenerate: (topic?: string) => void
   357→   357→  inputRef: React.RefObject<HTMLTextAreaElement | null>
   358→   358→}) {
   359→   359→  return (
   360→   360→    <GlassCard className="overflow-hidden" hover={false} gradient>
   361→   361→      <div className="p-5 sm:p-6">
   362→   362→        <label className="mb-2 block text-sm font-medium">Что хочешь изучить?</label>
   363→   363→        <textarea
   364→   364→          ref={inputRef}
   365→   365→          value={topic}
   366→   366→          onChange={(e) => setTopic(e.target.value)}
   367→   367→          placeholder="Например: как работают нейронные сети"
   368→   368→          rows={2}
   369→   369→          onKeyDown={(e) => {
   370→   370→            if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   371→   371→              e.preventDefault()
   372→   372→              onGenerate()
   373→   373→            }
   374→   374→          }}
   375→   375→          className="w-full resize-none rounded-xl border border-border bg-background/60 px-4 py-3 text-base outline-none ring-primary/20 transition placeholder:text-muted-foreground/60 focus:ring-2"
   376→   376→        />
   377→   377→
   378→   378→        {/* Subject chips */}
   379→   379→        <div className="mt-4">
   380→   380→          <p className="mb-2 text-xs text-muted-foreground">Предмет:</p>
   381→   381→          <div className="flex flex-wrap gap-2">
   382→   382→            <SubjectChip
   383→   383→              active={!subject}
   384→   384→              emoji="✨"
   385→   385→              label="Авто"
   386→   386→              onClick={() => setSubject(null)}
   387→   387→            />
   388→   388→            {SUBJECTS.map((s) => (
   389→   389→              <SubjectChip
   390→   390→                key={s.id}
   391→   391→                active={subject === s.id}
   392→   392→                emoji={s.emoji}
   393→   393→                label={s.ru}
   394→   394→                onClick={() => setSubject(s.id)}
   395→   395→              />
   396→   396→            ))}
   397→   397→          </div>
   398→   398→        </div>
   399→   399→
   400→   400→        {/* Level + Generate */}
   401→   401→        <div className="mt-4 flex flex-wrap items-center justify-between gap-3">
   402→   402→          <div>
   403→   403→            <p className="mb-1.5 text-xs text-muted-foreground">Уровень:</p>
   404→   404→            <div className="inline-flex items-center gap-1 rounded-lg border border-border/60 bg-muted/30 p-1">
   405→   405→              {LEVELS.map((l) => (
   406→   406→                <button
   407→   407→                  key={l}
   408→   408→                  onClick={() => setLevel(l)}
   409→   409→                  className={`rounded-md px-3 py-1 text-xs font-medium transition-colors ${
   410→   410→                    level === l
   411→   411→                      ? 'bg-primary text-primary-foreground shadow-sm'
   412→   412→                      : 'text-muted-foreground hover:text-foreground'
   413→   413→                  }`}
   414→   414→                >
   415→   415→                  {l}
   416→   416→                </button>
   417→   417→              ))}
   418→   418→            </div>
   419→   419→          </div>
   420→   420→          <GradientButton onClick={() => onGenerate()} disabled={loading} className="mt-5 sm:mt-0">
   421→   421→            {loading ? (
   422→   422→              <>
   423→   423→                <Loader2 className="h-4 w-4 animate-spin" /> Создаю урок...
   424→   424→              </>
   425→   425→            ) : (
   426→   426→              <>
   427→   427→                <Sparkles className="h-4 w-4" /> Создать урок
   428→   428→              </>
   429→   429→            )}
   430→   430→          </GradientButton>
   431→   431→        </div>
   432→   432→
   433→   433→        <p className="mt-3 text-[11px] text-muted-foreground/70">
   434→   434→          Подсказка: нажми <kbd className="rounded border border-border/60 bg-muted/40 px-1 py-0.5 font-mono text-[10px]">⌘/Ctrl + Enter</kbd> для быстрого запуска
   435→   435→        </p>
   436→   436→      </div>
   437→   437→    </GlassCard>
   438→   438→  )
   439→   439→}

   468→   468→
   469→   469→function CompactBar({
   470→   470→  lesson,
   471→   471→  onNew,
   472→   472→  onScrollToLesson,
   473→   473→  subjectLabel,
   474→   474→}: {
   475→   475→  lesson: Lesson
   476→   476→  onNew: () => void
   477→   477→  onScrollToLesson: () => void
   478→   478→  subjectLabel?: string
   479→   479→}) {
   480→   480→  return (
   481→   481→    <GlassCard className="overflow-hidden" hover={false}>
   482→   482→      <div className="flex flex-wrap items-center justify-between gap-3 p-3 sm:p-4">
   483→   483→        <button
   484→   484→          onClick={onScrollToLesson}
   485→   485→          className="flex min-w-0 flex-1 items-center gap-3 text-left"
   486→   486→        >
   487→   487→          <div className="grid h-10 w-10 shrink-0 place-items-center rounded-xl bg-gradient-to-br from-violet-500/20 to-fuchsia-500/20 text-xl ring-1 ring-primary/20">
   488→   488→            {lesson.emoji || '📖'}
   489→   489→          </div>
   490→   490→          <div className="min-w-0">
   491→   491→            <p className="truncate text-sm font-semibold">{lesson.title}</p>
   492→   492→            <p className="truncate text-xs text-muted-foreground">
   493→   493→              {subjectLabel ? `${subjectLabel} · ` : ''}
   494→   494→              {lesson.durationMin} мин · {lesson.difficulty}
